1. **Resumen del Proyecto:**

BetViero es un monorepo Python orientado a analitica de apuestas deportivas y registro operacional de picks. El repositorio agrupa tres dominios principales: League of Legends (`lol/`), tenis (`tennis/`) y futbol (`football/`), mas una aplicacion Streamlit raiz (`streamlit_app.py`) que actua como panel de control, dashboard de resultados, calculadora/registro de picks y lanzador de procesos para LoL y tenis. La capa persistente principal es Supabase/PostgREST, con esquemas SQL por dominio y credenciales leidas desde variables de entorno o `st.secrets`.

La arquitectura general combina pipelines batch, modelos ML y componentes de automatizacion. En LoL se ingieren datos historicos de Oracle's Elixir/PandaScore, se entrenan modelos de first blood y multimercado, se generan predicciones para partidos futuros, se calculan cuotas justas/EV/Kelly y se registran picks via Streamlit, Telegram y Supabase. En tenis se descargan datos historicos de Sackmann, se construyen features sin leakage, se consumen cuotas de The Odds API, se simulan partidos con Monte Carlo y se calculan value bets. En futbol hay un pipeline separado para API-Football/The Odds API con normalizacion de fixtures, estadisticas y entrenamiento/inferencia XGBoost.

El flujo operativo gira alrededor de archivos CSV/Parquet/modelos locales en `data/` y `models/`, scripts CLI en `scripts/`, modulos reutilizables en `src/`, y tablas Supabase para picks, historicos, odds y logs de apuestas. La app Streamlit importa directamente funciones de `lol.src.*` y `tennis.src.*`, por lo que concentra mucha logica de negocio, transformaciones de datos, acciones de control y renderizado UI en un unico archivo grande.

def filter_by_date(records: pd.DataFrame, start_date: date | None, end_date: date | None) -> pd.DataFrame:
    if records.empty:
        return records
    filtered = records.copy()
    created_dates = filtered["created_at"].dt.date
    if start_date is not None:
        filtered = filtered.loc[created_dates >= start_date]
        created_dates = filtered["created_at"].dt.date
    if end_date is not None:
        filtered = filtered.loc[created_dates <= end_date]
    return filtered.reset_index(drop=True)


def filter_by_investment_mode(records: pd.DataFrame, investment_mode: str) -> pd.DataFrame:
    if records.empty or investment_mode == "Combinado":
        return records
    target_bet_type = "paper" if investment_mode == "Paper" else "real"
    bet_types = records.get("bet_type", pd.Series("real", index=records.index)).fillna("real").astype(str).str.lower()
    return records.loc[bet_types.eq(target_bet_type)].reset_index(drop=True)

def save_predictions_snapshot(frame: pd.DataFrame, output_path: Path, merge: bool = True) -> pd.DataFrame:

def enrich_event_dates(records: pd.DataFrame, predictions_frame: pd.DataFrame) -> pd.DataFrame:
    if records.empty:
        return records
    enriched = records.copy()
    if "event_date" not in enriched.columns:
        enriched["event_date"] = pd.NaT
    enriched["event_date"] = pd.to_datetime(enriched["event_date"], utc=True, errors="coerce")
    if predictions_frame.empty or "match_name" not in predictions_frame.columns or "scheduled_at" not in predictions_frame.columns:
        return enriched
    schedule_map = (
        predictions_frame[["match_name", "scheduled_at"]]
        .dropna(subset=["match_name"])
        .drop_duplicates(subset=["match_name"], keep="last")
        .copy()
    )
    schedule_map["scheduled_at"] = pd.to_datetime(schedule_map["scheduled_at"], utc=True, errors="coerce")
    merged = enriched.merge(schedule_map, on="match_name", how="left")
    merged["event_date"] = merged["event_date"].fillna(merged["scheduled_at"])
    merged = merged.drop(columns=["scheduled_at"])
    return merged


def parse_manual_matches_text(text: str) -> pd.DataFrame:
    rows: list[dict[str, str]] = []
    for raw_line in text.splitlines():
        line = raw_line.strip()
        if not line:
            continue
        if "|" in line:
            league_part, match_part = [part.strip() for part in line.split("|", 1)]
            if " vs " not in match_part:
                raise LoLPipelineError(f"Linea invalida: {line}")
            team_a, team_b = [part.strip() for part in match_part.split(" vs ", 1)]
            rows.append({"league": league_part, "team_a": team_a, "team_b": team_b})
        else:
            parts = [part.strip() for part in line.split(",")]
            if len(parts) < 3:
                raise LoLPipelineError(f"Linea invalida: {line}")
            rows.append({"league": parts[0], "team_a": parts[1], "team_b": parts[2]})
    if not rows:
        raise LoLPipelineError("No se detectaron partidos manuales.")
    return pd.DataFrame(rows)


def build_valid_team_list(predictions_frame: pd.DataFrame) -> list[str]:
    if predictions_frame.empty:
        return []
    team_values: list[str] = []
    for column in ["team_a", "team_b"]:
        if column in predictions_frame.columns:
            team_values.extend(predictions_frame[column].dropna().astype(str).tolist())
    cleaned_teams = [apply_ocr_team_corrections(team_name) for team_name in team_values]
    cleaned_teams = [team_name for team_name in cleaned_teams if team_name]
    return sorted(set(cleaned_teams))


def find_prediction_match_name(predictions_frame: pd.DataFrame, team_a: str, team_b: str) -> str | None:
    if predictions_frame.empty or not {"team_a", "team_b", "match_name"}.issubset(predictions_frame.columns):
        return None
    target_pair = {clean_ocr_team_text(team_a).lower(), clean_ocr_team_text(team_b).lower()}
    for _, row in predictions_frame.iterrows():
        row_pair = {
            clean_ocr_team_text(str(row["team_a"])).lower(),
            clean_ocr_team_text(str(row["team_b"])).lower(),
        }
        if row_pair == target_pair:
            return str(row["match_name"])
    return None

        valid_teams = build_valid_team_list(predictions_frame)
        st.write(f"Debug equipos validos para fuzzy: {len(valid_teams)}")
        st.write("Muestra de equipos validos:", valid_teams[:5])
        print("Equipos validos para fuzzy:", valid_teams)
        if not valid_teams:
            st.warning("No hay equipos validos en el snapshot actual para hacer fuzzy matching.")
            return

        raw_matchups: list[str] = []
        for uploaded_image in uploaded_images:
            try:
                raw_matchups.extend(extract_matchups_from_image(uploaded_image.getvalue()))
            except Exception as exc:
                st.error(f"No pude procesar `{uploaded_image.name}`: {exc}")

        if not raw_matchups:
            st.warning("No se detectaron lineas de equipos en las capturas subidas.")
            return

        with st.expander("Ver texto bruto detectado (OCR Raw)"):
            st.write(raw_matchups)

        raw_matchups = list(dict.fromkeys(raw_matchups))
        detected_matchups = detect_matchups_from_ocr_lines(raw_matchups, valid_teams, minimum_score=70)
        if not detected_matchups:
            st.warning("No hay equipos con confianza alta. Revisa el OCR Raw o corrige manualmente en el siguiente ajuste.")
            return

        st.caption(f"OCR procesado sobre {len(uploaded_images)} captura(s). Valida los emparejamientos antes de continuar.")
        selector_options = ["Desconocido (Seleccionar...)"] + valid_teams
        match_options: list[tuple[str, str]] = []
        for row_index, detected_matchup in enumerate(detected_matchups):
            raw_line = str(detected_matchup["raw_line"])
            team_a = str(detected_matchup["team_a"])
            team_b = str(detected_matchup["team_b"])
            score_a = int(detected_matchup["score_a"])
            score_b = int(detected_matchup["score_b"])

            st.markdown(f"**Linea OCR:** {raw_line}")
            col_detected_a, col_confidence_a, col_selector_a = st.columns([2, 1, 3])
            col_detected_a.write(team_a)
            col_confidence_a.write(f"{score_a}%")
            default_a = selector_options.index(team_a) if team_a in selector_options else 0
            selected_team_a = col_selector_a.selectbox(
                "Selector equipo 1",
                selector_options,
                index=default_a,
                key=f"magic_team_a_selector_{row_index}_{team_a}",
            )

            col_detected_b, col_confidence_b, col_selector_b = st.columns([2, 1, 3])
            col_detected_b.write(team_b or "Desconocido")
            col_confidence_b.write(f"{score_b}%")
            default_b = selector_options.index(team_b) if team_b in selector_options else 0
            selected_team_b = col_selector_b.selectbox(
                "Selector equipo 2",
                selector_options,
                index=default_b,
                key=f"magic_team_b_selector_{row_index}_{team_b or 'unknown'}",
            )

            if selected_team_a.startswith("Desconocido") or selected_team_b.startswith("Desconocido"):
                continue

            team_a = selected_team_a
            team_b = selected_team_b
            match_name = find_prediction_match_name(predictions_frame, team_a, team_b)
            label = f"{team_a} vs {team_b}"
            if match_name is None:
                label = f"{label} (no encontrado en snapshot)"
            match_options.append((label, match_name or ""))

        if not match_options:
            st.info("Selecciona al menos dos equipos para formar un partido.")
            return

        labels = [label for label, _ in match_options]
        selected_label = st.selectbox("Partido detectado para cargar", labels)
        selected_match_name = dict(match_options).get(selected_label)
        if st.button("Cargar en Calculadora", use_container_width=True):
            if not selected_match_name:
                st.warning("Ese partido no existe en el snapshot actual. Primero genera predicciones para ese matchup.")
            else:
                st.session_state["register_match_name"] = selected_match_name
                st.session_state["lol_register_match_name"] = selected_match_name
                st.session_state["force_register_page"] = True
                st.success(f"Partido cargado en calculadora: {selected_match_name}")
                st.rerun()


def compute_pick_payload_from_row(
    config: BotConfig,
    row: pd.Series,
    market: str,
    selection_key: str,
    my_odd: float,
    bankroll_amount: float,
    line: float | None = None,
) -> dict[str, object]:
    if market in {"winner", "firstblood", "firsttower"}:
        if market == "firstblood" and str(row["league"]) == "LPL":
            raise LoLPipelineError("First Blood queda desactivado para LPL.")
        team_name = str(row["team_a"]) if selection_key == "1" else str(row["team_b"])
        probability = float(row[f"{market}_{'team_a' if selection_key == '1' else 'team_b'}_prob"])
        fair_odd = float(row[f"{market}_{'team_a' if selection_key == '1' else 'team_b'}_fair_odd"])
        selection = team_name
        predicted_value = None
    else:
        if line is None:
            raise LoLPipelineError("La linea es obligatoria para kills.")
        probability = estimate_over_under_probability(
            predicted_total=float(row["kills_prediction"]),
            line=float(line),
            residual_std=float(row.get("kills_residual_std", 6.0)),
            direction=selection_key,
        )
        fair_odd = calculate_fair_odd(probability)
        selection = selection_key
        predicted_value = float(row["kills_prediction"])

    implied_probability = calculate_implied_probability(my_odd)
    edge = probability - implied_probability
    ev = calculate_ev(probability, my_odd)
    full_kelly = calculate_kelly_fraction(probability, my_odd)
    recommended_stake = min(
        bankroll_amount * full_kelly * config.kelly_fraction,
        config.max_stake_amount,
    )
    return {
        "match_name": str(row["match_name"]),
        "league": str(row["league"]),
        "market": market,
        "selection": selection,
        "line": line,
        "my_odd": my_odd,
        "model_probability": probability,
        "model_fair_odd": fair_odd,
        "predicted_value": predicted_value,
        "implied_probability": implied_probability,
        "edge": edge,
        "ev": ev,
        "full_kelly": full_kelly,
        "kelly_fraction": config.kelly_fraction,
        "recommended_stake": recommended_stake,
        "bankroll_snapshot": bankroll_amount,
        "status": "pending",
    }


def render_overview(records: pd.DataFrame) -> None:
    st.subheader("Pendientes")
    pending_view = records.loc[
        records["status"].eq("pending"),
        [
            "created_at",
            "source_table",
            "bet_type",
            "league",
            "match_name",
            "market",
            "selection",
            "line",
            "my_odd",
            "model_fair_odd",
            "recommended_stake",
            "edge",
            "ev",
        ],
    ].copy()
    st.dataframe(pending_view, use_container_width=True, hide_index=True)

    st.subheader("Desgloses")
    left, right = st.columns(2)
    with left:
        by_market = (
            records.groupby("market", dropna=False)
            .agg(records=("record_id", "count"), avg_edge=("edge", "mean"), avg_ev=("ev", "mean"), pnl=("pnl", "sum"))
            .reset_index()
        )
        st.dataframe(by_market, use_container_width=True, hide_index=True)
    with right:
        by_status = (
            records.groupby("status", dropna=False)
            .agg(records=("record_id", "count"), pnl=("pnl", "sum"))
            .reset_index()
        )
        st.dataframe(by_status, use_container_width=True, hide_index=True)

def style_tennis_value_rows(row: pd.Series) -> list[str]:
    ev_value = pd.to_numeric(row.get("EV Actual", row.get("EV", 0)), errors="coerce")
    if pd.notna(ev_value) and float(ev_value) > 0:
        return ["background-color: #d8f5df; color: #102a18;" for _ in row]
    return ["" for _ in row]


def add_tennis_kelly_column(frame: pd.DataFrame, bankroll: float = 1000.0) -> pd.DataFrame:
    if frame.empty or not {"EV", "cuota_casa"}.issubset(frame.columns):
        return frame
    output = frame.copy()
    ev_values = pd.to_numeric(output["EV"], errors="coerce")
    odds_values = pd.to_numeric(output["cuota_casa"], errors="coerce")
    denominator = odds_values - 1
    output["Kelly Recomendado"] = (bankroll * ev_values / denominator).where(denominator > 0)
    return output


def tennis_file_status(path: Path) -> dict[str, object]:
    status: dict[str, object] = {
        "archivo": str(path.relative_to(PROJECT_ROOT)),
        "existe": path.exists(),
        "bytes": path.stat().st_size if path.exists() else 0,
    }
    if path.exists() and path.suffix.lower() == ".csv":
        try:
            status["filas"] = len(pd.read_csv(path))
        except Exception as exc:
            status["filas"] = f"Error leyendo CSV: {exc}"
    return status

def settle_tennis_bet(bet_id: str, status: str, result_selection: str | None = None) -> None:
    bets = load_tennis_bets_log()
    if bets.empty:
        raise ValueError("No hay apuestas cargadas.")

    row = bets.loc[bets["id"].eq(bet_id)]
    if row.empty:
        raise ValueError("No encontre la apuesta seleccionada.")

    bet = row.iloc[0]
    stake = float(bet["stake"])
    odd = float(bet["my_odd"])
    if status == "won":
        pnl = stake * (odd - 1.0)
    elif status == "lost":
        pnl = -stake
    elif status == "void":
        pnl = 0.0
    else:
        raise ValueError(f"Estado no soportado: {status}")

    payload = {
        "status": status,
        "result_selection": result_selection or None,
        "pnl": pnl,
        "settled_at": pd.Timestamp.utcnow().isoformat(),
    }
    update_tennis_row("bets_log", bet_id, payload)


def build_tennis_best_market_odds(frame: pd.DataFrame) -> pd.DataFrame:
    if frame.empty:
        return pd.DataFrame()

    sort_columns = ["external_event_id", "market_key", "selection", "line", "price"]
    best_rows = (
        frame.sort_values(sort_columns, ascending=[True, True, True, True, False], na_position="first")
        .drop_duplicates(subset=["external_event_id", "market_key", "selection", "line"], keep="first")
        .copy()
    )
    grouped = frame.groupby(["external_event_id", "market_key", "selection", "line"], dropna=False)
    summary = grouped.agg(
        casas=("bookmaker_key", "nunique"),
        min_cuota=("price", "min"),
        max_cuota=("price", "max"),
        ultima_actualizacion=("collected_at", "max"),
    ).reset_index()

    output = best_rows.merge(
        summary,
        on=["external_event_id", "market_key", "selection", "line"],
        how="left",
        suffixes=("", "_summary"),
    )
    output["rango_cuotas"] = output["max_cuota"] - output["min_cuota"]
    return output.sort_values(["commence_time", "match_label", "market_key", "selection", "line"]).reset_index(drop=True)


def parse_tennis_value_market(mercado: object) -> tuple[str | None, str | None]:
    market_text = "" if pd.isna(mercado) else str(mercado).strip()
    lowered = market_text.lower()
    if lowered == "over":
        return "totals", "Over"
    if lowered == "under":
        return "totals", "Under"
    if lowered.startswith("handicap "):
        return "spreads", market_text[len("Handicap ") :].strip()
    if market_text:
        return "h2h", market_text
    return None, None


def enrich_tennis_value_bets_with_market(value_bets: pd.DataFrame, bankroll: float = 1000.0) -> pd.DataFrame:
    if value_bets.empty:
        return value_bets

    output = value_bets.copy()
    if "linea" in output.columns:
        output["linea"] = pd.to_numeric(output["linea"], errors="coerce")
    if "nuestra_prob" in output.columns:
        output["nuestra_prob"] = pd.to_numeric(output["nuestra_prob"], errors="coerce")

    parsed_markets = output["mercado"].map(parse_tennis_value_market)
    output["market_key_actual"] = parsed_markets.map(lambda value: value[0])
    output["selection_actual"] = parsed_markets.map(lambda value: value[1])

    try:
        picks = load_tennis_picks_history()
    except Exception:
        picks = pd.DataFrame()
    if not picks.empty:
        picks = picks.copy()
        picks["linea"] = pd.to_numeric(picks["linea"], errors="coerce")
        picks["fecha_registro"] = pd.to_datetime(picks["fecha_registro"], errors="coerce")
        latest_picks = (
            picks.sort_values("fecha_registro", ascending=False)
            .drop_duplicates(subset=["partido", "mercado", "linea"], keep="first")
            [["id", "partido", "mercado", "linea", "fecha_registro"]]
            .rename(columns={"id": "tennis_pick_id", "fecha_registro": "tennis_pick_fecha"})
        )
        output = output.merge(
            latest_picks,
            on=["partido", "mercado", "linea"],
            how="left",
        )

    try:
        market_odds = load_tennis_upcoming_odds()
    except Exception:
        return output
    if market_odds.empty:
        return output

    best_odds = build_tennis_best_market_odds(market_odds)
    if best_odds.empty:
        return output

    join_columns_left = ["partido", "market_key_actual", "selection_actual", "linea"]
    join_columns_right = ["match_label", "market_key", "selection", "line"]
    current_odds = best_odds[
        [
            "match_label",
            "market_key",
            "selection",
            "line",
            "price",
            "bookmaker_key",
            "bookmaker_title",
            "casas",
            "min_cuota",
            "max_cuota",
            "rango_cuotas",
            "ultima_actualizacion",
        ]
    ].rename(
        columns={
            "price": "mejor_cuota_actual",
            "bookmaker_key": "mejor_bookmaker_key_actual",
            "bookmaker_title": "mejor_casa_actual",
            "casas": "casas_actuales",
            "min_cuota": "min_cuota_actual",
            "max_cuota": "max_cuota_actual",
            "rango_cuotas": "rango_cuotas_actual",
            "ultima_actualizacion": "ultima_actualizacion_cuota",
        }
    )

    output = output.merge(
        current_odds,
        left_on=join_columns_left,
        right_on=join_columns_right,
        how="left",
    )
    output["delta_cuota_actual"] = output["mejor_cuota_actual"] - pd.to_numeric(output.get("cuota_casa"), errors="coerce")
    output["EV Actual"] = output["nuestra_prob"] * output["mejor_cuota_actual"] - 1.0
    denominator = output["mejor_cuota_actual"] - 1.0
    output["Kelly Actual"] = (bankroll * output["EV Actual"] / denominator).where(denominator > 0)
    return output


def split_tennis_match_name(match_name: object) -> tuple[str, str]:
    text = "" if pd.isna(match_name) else str(match_name)
    if " vs " not in text:
        return text.strip(), ""
    player_1, player_2 = text.split(" vs ", 1)
    return player_1.strip(), player_2.strip()


def calculate_tennis_kelly_values(probability: float, decimal_odd: float, bankroll: float, multiplier: float) -> tuple[float, float, float]:
    ev = probability * decimal_odd - 1.0
    full_kelly_fraction = ev / (decimal_odd - 1.0) if decimal_odd > 1.0 else 0.0
    applied_fraction = max(0.0, full_kelly_fraction * multiplier)
    stake = bankroll * applied_fraction
    return ev, applied_fraction, stake


def insert_tennis_bet_log(row: pd.Series, *, bet_type: str, odd: float, stake: float, kelly_fraction: float, notes: str) -> None:
    player_1, player_2 = split_tennis_match_name(row.get("partido"))
    market_key, selection = parse_tennis_value_market(row.get("mercado"))
    probability = float(row.get("nuestra_prob"))
    ev = probability * odd - 1.0
    payload = {
        "tennis_pick_id": None if pd.isna(row.get("tennis_pick_id")) else str(row.get("tennis_pick_id")),
        "source": "streamlit_value_bet",
        "bet_type": bet_type,
        "player_1": player_1,
        "player_2": player_2,
        "market_key": market_key or str(row.get("mercado") or ""),
        "selection": selection or str(row.get("mercado") or ""),
        "line": None if pd.isna(row.get("linea")) else float(row.get("linea")),
        "bookmaker_key": str(row.get("mejor_bookmaker_key_actual") or row.get("mejor_casa_actual") or row.get("casa_apuestas") or ""),
        "my_odd": float(odd),
        "model_probability": probability,
        "model_fair_odd": 1.0 / probability if probability > 0 else None,
        "edge": probability - (1.0 / odd),
        "ev": ev,
        "stake": float(stake),
        "kelly_fraction": float(kelly_fraction),
        "status": "pending",
        "notes": notes.strip() or None,
        "raw_payload": {
            "partido": row.get("partido"),
            "mercado": row.get("mercado"),
            "cuota_casa_csv": None if pd.isna(row.get("cuota_casa")) else float(row.get("cuota_casa")),
            "casa_apuestas_csv": row.get("casa_apuestas"),
            "mejor_cuota_actual": None if pd.isna(row.get("mejor_cuota_actual")) else float(row.get("mejor_cuota_actual")),
            "mejor_casa_actual": row.get("mejor_casa_actual"),
            "tennis_pick_id": None if pd.isna(row.get("tennis_pick_id")) else str(row.get("tennis_pick_id")),
            "tennis_pick_fecha": None if pd.isna(row.get("tennis_pick_fecha")) else str(row.get("tennis_pick_fecha")),
            "ev_csv": None if pd.isna(row.get("EV")) else float(row.get("EV")),
            "ev_actual": None if pd.isna(row.get("EV Actual")) else float(row.get("EV Actual")),
        },
    }
    (
        get_supabase_client()
        .schema("tennis")
        .table("bets_log")
        .insert(payload)
        .execute()
    )


def build_tennis_odds_tracking_frame() -> pd.DataFrame:
    picks = load_tennis_picks_history()
    history = load_tennis_pick_odds_history()
    if picks.empty or history.empty:
        return pd.DataFrame()

    tracking = history.merge(
        picks,
        left_on="pick_id",
        right_on="id",
        how="left",
        suffixes=("_snapshot", "_pick"),
    )
    if tracking.empty:
        return tracking

    for column in ["cuota", "ev_momento", "linea", "cuota_inicial", "nuestra_prob", "ev_inicial"]:
        if column in tracking.columns:
            tracking[column] = pd.to_numeric(tracking[column], errors="coerce")

    for column in ["fecha_registro_snapshot", "fecha_registro_pick"]:
        if column in tracking.columns:
            tracking[column] = pd.to_datetime(tracking[column], errors="coerce")

    tracking["linea_label"] = tracking["linea"].map(lambda value: "-" if pd.isna(value) else f"{value:g}")
    tracking["pick_label"] = (
        tracking["partido"].fillna("Sin partido")
        + " | "
        + tracking["mercado"].fillna("Sin mercado")
        + " "
        + tracking["linea_label"]
        + " | "
        + tracking["casa_apuestas"].fillna("Sin casa")
    )
    return tracking.sort_values(["pick_label", "fecha_registro_snapshot"]).reset_index(drop=True)

def render_tennis_bet_registration(value_bets: pd.DataFrame) -> None:
    st.subheader("Registrar apuesta")
    candidate_bets = value_bets.copy()
    if "EV Actual" in candidate_bets.columns:
        candidate_bets["ev_para_registro"] = pd.to_numeric(candidate_bets["EV Actual"], errors="coerce")
    else:
        candidate_bets["ev_para_registro"] = pd.to_numeric(candidate_bets.get("EV"), errors="coerce")
    candidate_bets = candidate_bets.loc[candidate_bets["ev_para_registro"].fillna(-1) > 0].copy()
    if candidate_bets.empty:
        st.info("No hay value bets con EV actual positivo para registrar.")
        return

    candidate_bets = candidate_bets.reset_index(drop=True)
    candidate_bets["registro_label"] = candidate_bets.apply(
        lambda row: (
            f"{int(row.name) + 1}. {row.get('partido', '-')} | {row.get('mercado', '-')} "
            f"{row.get('linea', '-')} | cuota {row.get('mejor_cuota_actual', row.get('cuota_casa', '-'))}"
        ),
        axis=1,
    )
    selected_label = st.selectbox("Value bet", candidate_bets["registro_label"].tolist())
    selected_row = candidate_bets.loc[candidate_bets["registro_label"].eq(selected_label)].iloc[0]

    default_odd = selected_row.get("mejor_cuota_actual")
    if pd.isna(default_odd):
        default_odd = selected_row.get("cuota_casa")
    default_odd = float(default_odd)
    probability = float(selected_row.get("nuestra_prob"))

    col_1, col_2, col_3, col_4 = st.columns(4)
    col_1.metric("Prob modelo", f"{probability:.2%}")
    col_2.metric("Cuota por defecto", f"{default_odd:.2f}")
    col_3.metric("Casa", str(selected_row.get("mejor_casa_actual") or selected_row.get("casa_apuestas") or "-"))
    col_4.metric("EV actual", f"{(probability * default_odd - 1.0):.2%}")

    form_col_1, form_col_2, form_col_3, form_col_4 = st.columns(4)
    bet_type = form_col_1.selectbox("Tipo", ["real", "paper"])
    bankroll = form_col_2.number_input("Bankroll", min_value=0.0, value=1000.0, step=50.0)
    kelly_multiplier = form_col_3.selectbox("Kelly", [0.25, 0.5, 1.0], format_func=lambda value: f"{value:g}x")
    manual_odd = form_col_4.number_input("Cuota", min_value=1.01, value=default_odd, step=0.01, format="%.2f")

    ev, applied_kelly_fraction, recommended_stake = calculate_tennis_kelly_values(
        probability,
        manual_odd,
        bankroll,
        kelly_multiplier,
    )
    use_recommended_stake = st.checkbox("Usar stake Kelly recomendado", value=True)
    if use_recommended_stake:
        stake = float(round(recommended_stake, 2))
        st.metric("Stake a registrar", f"{stake:.2f}")
    else:
        stake = st.number_input("Stake manual", min_value=0.0, value=float(round(recommended_stake, 2)), step=1.0, format="%.2f")
    notes = st.text_area("Notas", value="")

    calc_col_1, calc_col_2, calc_col_3 = st.columns(3)
    calc_col_1.metric("EV con tu cuota", f"{ev:.2%}")
    calc_col_2.metric("Kelly aplicado", f"{applied_kelly_fraction:.2%}")
    calc_col_3.metric("Stake recomendado", f"{recommended_stake:.2f}")

    if st.button("Registrar en bets_log", use_container_width=True):
        try:
            stored_kelly_fraction = stake / bankroll if bankroll > 0 else applied_kelly_fraction
            insert_tennis_bet_log(
                selected_row,
                bet_type=bet_type,
                odd=manual_odd,
                stake=stake,
                kelly_fraction=stored_kelly_fraction,
                notes=notes,
            )
            load_tennis_bets_log.clear()
            st.success("Apuesta registrada en tennis.bets_log.")
        except Exception as exc:
            st.error(f"No pude registrar la apuesta: {exc}")

def render_tennis_odds_tracking() -> None:

    try:
        tracking = build_tennis_odds_tracking_frame()

        st.error(f"No pude cargar el tracking de cuotas: {exc}")
        return

    if tracking.empty:
        st.info("No hay historico de cuotas para graficar todavia.")
        return

    tracked_pick_count = int(tracking["pick_id"].nunique())
    snapshot_count = len(tracking)
    metric_1, metric_2, metric_3 = st.columns(3)
    metric_1.metric("Picks trackeados", tracked_pick_count)
    metric_2.metric("Snapshots", snapshot_count)
    metric_3.metric("Media snapshots/pick", f"{snapshot_count / max(tracked_pick_count, 1):.1f}")

    pick_options = tracking["pick_label"].dropna().drop_duplicates().sort_values().tolist()
    selected_pick = st.selectbox("Pick", pick_options)

    if chart_frame.empty:
        st.info("No hay puntos para el pick seleccionado.")
        return

    selected_summary = chart_frame.groupby("pick_label", as_index=False).agg(
        primera_cuota=("cuota", "first"),
        ultima_cuota=("cuota", "last"),
        max_cuota=("cuota", "max"),
        min_cuota=("cuota", "min"),
        snapshots=("cuota", "count"),
    )
    selected_summary["movimiento"] = selected_summary["ultima_cuota"] - selected_summary["primera_cuota"]

    if not show_all and not selected_summary.empty:
        row = selected_summary.iloc[0]
        col_1, col_2, col_3, col_4 = st.columns(4)
        col_1.metric("Cuota inicial", f"{row['primera_cuota']:.2f}")
        col_2.metric("Cuota actual", f"{row['ultima_cuota']:.2f}", delta=f"{row['movimiento']:.2f}")
        col_3.metric("Max", f"{row['max_cuota']:.2f}")
        col_4.metric("Min", f"{row['min_cuota']:.2f}")

    st.subheader("Evolucion de cuota")
    st.line_chart(
        chart_frame,
        x="fecha_registro_snapshot",
        y="cuota",
        color="pick_label" if show_all else None,
        use_container_width=True,
    )

    if "ev_momento" in chart_frame.columns and chart_frame["ev_momento"].notna().any():
        st.subheader("Evolucion de EV")
        st.line_chart(
            chart_frame,
            x="fecha_registro_snapshot",
            y="ev_momento",
            color="pick_label" if show_all else None,
            use_container_width=True,
        )

    st.subheader("Ultimos puntos")
    table_columns = [
        "fecha_registro_snapshot",
        "partido",
        "mercado",
        "linea",
        "casa_apuestas",
        "cuota",
        "ev_momento",
    ]
    visible_columns = [column for column in table_columns if column in chart_frame.columns]
    st.dataframe(
        chart_frame.sort_values("fecha_registro_snapshot", ascending=False)[visible_columns].head(100),
        use_container_width=True,
        hide_index=True,
        column_config={
            "fecha_registro_snapshot": st.column_config.DatetimeColumn("Fecha", format="YYYY-MM-DD HH:mm"),
            "linea": st.column_config.NumberColumn("Linea", format="%.1f"),

        },

def render_tennis_bets_log() -> None:

        load_tennis_bets_log.clear()

    try:
        bets = load_tennis_bets_log()
    except Exception as exc:
        st.error(f"No pude cargar bets_log: {exc}")
        return

    if bets.empty:
        st.info("No hay apuestas registradas todavia.")
        return

    settled_bets = bets.loc[bets["status"].isin(["won", "lost", "void"])].copy()
    real_bets = bets.loc[bets["bet_type"].eq("real")].copy()
    pending_bets = bets.loc[bets["status"].eq("pending")].copy()
    total_staked = float(bets["stake"].sum())
    total_pnl = float(settled_bets["pnl"].fillna(0).sum()) if "pnl" in settled_bets.columns else 0.0
    real_pnl = float(real_bets["pnl"].fillna(0).sum()) if "pnl" in real_bets.columns else 0.0
    roi = total_pnl / total_staked if total_staked > 0 else 0.0

    metric_1, metric_2, metric_3, metric_4, metric_5 = st.columns(5)
    metric_1.metric("Apuestas", len(bets))
    metric_2.metric("Pendientes", len(pending_bets))

    metric_4.metric("PnL total", f"{total_pnl:.2f}")
    metric_5.metric("ROI", f"{roi:.2%}")
    st.caption(f"PnL real: {real_pnl:.2f}")

    type_options = ["Todos"] + bets["bet_type"].dropna().drop_duplicates().sort_values().tolist()
    status_options = ["Todos"] + bets["status"].dropna().drop_duplicates().sort_values().tolist()
    col_type, col_status = st.columns(2)
    selected_type = col_type.selectbox("Tipo", type_options)
    selected_status = col_status.selectbox("Estado", status_options)

    if selected_type != "Todos":
        filtered = filtered.loc[filtered["bet_type"].eq(selected_type)]
    if selected_status != "Todos":
        filtered = filtered.loc[filtered["status"].eq(selected_status)]

    visible_columns = [
        "placed_at",
        "tennis_pick_id",
        "bet_type",
        "status",
        "partido",
        "market_key",
        "selection",
        "line",
        "bookmaker_key",
        "my_odd",
        "model_probability",
        "ev",
        "stake",
        "kelly_fraction",
        "pnl",
        "settled_at",
        "notes",
    ]
    st.dataframe(
        filtered[[column for column in visible_columns if column in filtered.columns]],
        use_container_width=True,
        hide_index=True,
        column_config={
            "placed_at": st.column_config.DatetimeColumn("Fecha", format="YYYY-MM-DD HH:mm"),
            "tennis_pick_id": "Pick ID",
            "bet_type": "Tipo",
            "status": "Estado",
            "partido": "Partido",
            "market_key": "Mercado",
            "selection": "Seleccion",
            "line": st.column_config.NumberColumn("Linea", format="%.1f"),
            "bookmaker_key": "Casa",
            "my_odd": st.column_config.NumberColumn("Cuota", format="%.2f"),
            "model_probability": st.column_config.NumberColumn("Prob modelo", format="%.2f"),
            "ev": st.column_config.NumberColumn("EV", format="%.2f"),
            "stake": st.column_config.NumberColumn("Stake", format="%.2f"),
            "kelly_fraction": st.column_config.NumberColumn("Kelly", format="%.2f"),
            "pnl": st.column_config.NumberColumn("PnL", format="%.2f"),
            "settled_at": st.column_config.DatetimeColumn("Cierre", format="YYYY-MM-DD HH:mm"),
        },
    )

    st.subheader("Cerrar apuesta pendiente")
    if pending_bets.empty:
        st.info("No hay apuestas pendientes.")
        return

    selected_bet_label = st.selectbox("Apuesta pendiente", pending_bets["bet_label"].tolist())
    selected_bet = pending_bets.loc[pending_bets["bet_label"].eq(selected_bet_label)].iloc[0]
    settle_col_1, settle_col_2 = st.columns(2)
    settle_status = settle_col_1.selectbox("Resultado", ["won", "lost", "void"])
    default_result = str(selected_bet.get("selection") or "") if settle_status == "won" else ""
    result_selection = settle_col_2.text_input("Seleccion ganadora / resultado", value=default_result)

    projected_pnl = 0.0
    if settle_status == "won":
        projected_pnl = float(selected_bet["stake"]) * (float(selected_bet["my_odd"]) - 1.0)
    elif settle_status == "lost":
        projected_pnl = -float(selected_bet["stake"])
    st.metric("PnL proyectado", f"{projected_pnl:.2f}")

    if st.button("Cerrar apuesta", use_container_width=True):
        try:
            settle_tennis_bet(str(selected_bet["id"]), settle_status, result_selection)
            load_tennis_bets_log.clear()

            st.rerun()

class FairOddsEngine:
    def __init__(self, config: BotConfig) -> None:
        self.config = config
        self.overrides = load_name_overrides(config.team_name_overrides_path)
        self.feature_store = HistoricalFeatureStore(config.historical_team_rows_path, self.overrides)
        with config.model_artifact_path.open("rb") as file_obj:
            self.artifact = pickle.load(file_obj)
        self.market_models = self.artifact["models"]

    def fetch_upcoming_matches(self) -> list[dict[str, Any]]:
        headers = {"Authorization": f"Bearer {self.config.pandascore_api_key}"}
        matches: list[dict[str, Any]] = []
        page = 1
        while page <= 5:
            response = requests.get(
                f"{self.config.pandascore_base_url}/lol/matches/upcoming",
                headers=headers,
                params={"page": page, "per_page": 100},
                timeout=30,
            )
            response.raise_for_status()
            payload = response.json()
            if not payload:
                break
            matches.extend(payload)
            if len(payload) < 100:
                break
            page += 1
        return [match for match in matches if is_major_league(match)]

    def _extract_match_datetime(self, match_payload: dict[str, Any]) -> datetime | None:
        for key in ("begin_at", "scheduled_at", "original_scheduled_at"):
            raw_value = match_payload.get(key)
            if raw_value:
                return datetime.fromisoformat(str(raw_value).replace("Z", "+00:00"))
        return None

    def _extract_patch(self, match_payload: dict[str, Any], league_name: str) -> str:
        for key in ("videogame_version", "version", "patch"):
            raw_value = match_payload.get(key)
            if isinstance(raw_value, dict):
                for nested_key in ("name", "slug"):
                    if raw_value.get(nested_key):
                        return self.feature_store.resolve_patch(league_name, raw_value[nested_key])
            if raw_value:
                return self.feature_store.resolve_patch(league_name, raw_value)
        return self.feature_store.resolve_patch(league_name, "")

    def _predict_blue_probability(self, market: str, matchup_row: dict[str, Any]) -> float:
        market_artifact = self.market_models[market]
        frame = pd.DataFrame([matchup_row])
        feature_columns = market_artifact["feature_columns"]
        missing_columns = [column for column in feature_columns if column not in frame.columns]
        if missing_columns:
            missing_list = ", ".join(missing_columns)
            raise LoLPipelineError(f"Missing inference features for {market}: {missing_list}")
        probabilities = market_artifact["model"].predict_proba(frame[feature_columns])[:, 1]
        return float(probabilities[0])

    def _predict_total_kills(self, matchup_row: dict[str, Any]) -> float:
        market_artifact = self.market_models["kills"]
        frame = pd.DataFrame([matchup_row])
        feature_columns = market_artifact["feature_columns"]
        missing_columns = [column for column in feature_columns if column not in frame.columns]
        if missing_columns:
            missing_list = ", ".join(missing_columns)
            raise LoLPipelineError(f"Missing inference features for kills: {missing_list}")
        prediction = market_artifact["model"].predict(frame[feature_columns])
        return float(prediction[0])

    def score_upcoming_matches(self) -> pd.DataFrame:
        raw_matches = self.fetch_upcoming_matches()
        scored_rows: list[dict[str, Any]] = []
        horizon = datetime.now(UTC) + timedelta(hours=self.config.pandascore_lookahead_hours)

        for match_payload in raw_matches:
            opponents = match_payload.get("opponents") or []
            team_names = [
                str(opponent.get("opponent", {}).get("name", "")).strip()
                for opponent in opponents
                if opponent.get("opponent", {}).get("name")
            ]
            if len(team_names) != 2:
                continue

            scheduled_at = self._extract_match_datetime(match_payload)
            if scheduled_at is not None and scheduled_at > horizon:
                continue

            team_a, team_b = team_names[0], team_names[1]
            league_name = str((match_payload.get("league") or {}).get("name", "Unknown")).strip()
            patch = self._extract_patch(match_payload, league_name)
            match_name = f"{team_a} vs {team_b}"

            scenario_a_blue = self._build_matchup_row(
                blue_external_name=team_a,
                red_external_name=team_b,
                blue_has_firstpick=1.0,
                league_name=league_name,
                patch=patch,
                match_name=match_name,
                scheduled_at=scheduled_at,
                source_match_id=match_payload.get("id"),
            )
            scenario_a_blue_red_fp = self._build_matchup_row(
                blue_external_name=team_a,
                red_external_name=team_b,
                blue_has_firstpick=0.0,
                league_name=league_name,
                patch=patch,
                match_name=match_name,
                scheduled_at=scheduled_at,
                source_match_id=match_payload.get("id"),
            )
            scenario_b_blue = self._build_matchup_row(
                blue_external_name=team_b,
                red_external_name=team_a,
                blue_has_firstpick=1.0,
                league_name=league_name,
                patch=patch,
                match_name=match_name,
                scheduled_at=scheduled_at,
                source_match_id=match_payload.get("id"),
            )
            scenario_b_blue_red_fp = self._build_matchup_row(
                blue_external_name=team_b,
                red_external_name=team_a,
                blue_has_firstpick=0.0,
                league_name=league_name,
                patch=patch,
                match_name=match_name,
                scheduled_at=scheduled_at,
                source_match_id=match_payload.get("id"),
            )

            row_payload: dict[str, Any] = {
                "source_match_id": str(match_payload.get("id")),
                "match_name": match_name,
                "league": league_name,
                "patch": patch,
                "scheduled_at": scheduled_at.isoformat().replace("+00:00", "Z")
                if scheduled_at is not None
                else None,
                "team_a": team_a,
                "team_b": team_b,
                "side_assignment_method": "average_blue_red_scenarios",
            }

            for market in ["winner", "firstblood", "firsttower"]:
                team_a_if_blue = self._predict_blue_probability(market, scenario_a_blue)
                team_a_if_blue_red_fp = self._predict_blue_probability(market, scenario_a_blue_red_fp)
                team_a_if_red = 1.0 - self._predict_blue_probability(market, scenario_b_blue)
                team_a_if_red_red_fp = 1.0 - self._predict_blue_probability(
                    market,
                    scenario_b_blue_red_fp,
                )
                team_a_probability = (
                    team_a_if_blue
                    + team_a_if_blue_red_fp
                    + team_a_if_red
                    + team_a_if_red_red_fp
                ) / 4.0
                team_b_probability = 1.0 - team_a_probability
                row_payload[f"{market}_team_a_prob"] = team_a_probability
                row_payload[f"{market}_team_b_prob"] = team_b_probability
                row_payload[f"{market}_team_a_fair_odd"] = calculate_fair_odd(team_a_probability)
                row_payload[f"{market}_team_b_fair_odd"] = calculate_fair_odd(team_b_probability)

            kills_if_a_blue = self._predict_total_kills(scenario_a_blue)
            kills_if_a_blue_red_fp = self._predict_total_kills(scenario_a_blue_red_fp)
            kills_if_b_blue = self._predict_total_kills(scenario_b_blue)
            kills_if_b_blue_red_fp = self._predict_total_kills(scenario_b_blue_red_fp)
            row_payload["kills_prediction"] = (
                kills_if_a_blue
                + kills_if_a_blue_red_fp
                + kills_if_b_blue
                + kills_if_b_blue_red_fp
            ) / 4.0
            row_payload["kills_residual_std"] = float(
                self.market_models["kills"]["metrics"].get("residual_std", 6.0)
            )
            scored_rows.append(row_payload)

        output = pd.DataFrame(scored_rows)
        if not output.empty:
            output = output.sort_values(["scheduled_at", "league", "match_name"]).reset_index(drop=True)
            self.config.predictions_output_path.parent.mkdir(parents=True, exist_ok=True)
            output.to_csv(self.config.predictions_output_path, index=False)
        return output

def build_alert_message(match_row: pd.Series) -> str:
    team_a = str(match_row["team_a"])
    team_b = str(match_row["team_b"])
    league = str(match_row["league"])
    scheduled_at_text = ""
    scheduled_at = match_row.get("scheduled_at")
    if pd.notna(scheduled_at) and scheduled_at:
        scheduled_at_text = f"Hora UTC: {scheduled_at}\n"
    firstblood_line = ""
    if league != "LPL":
        firstblood_line = (
            f"First Blood: Equipo 1 @{float(match_row['firstblood_team_a_fair_odd']):.2f} | "
            f"Equipo 2 @{float(match_row['firstblood_team_b_fair_odd']):.2f}\n"
        )
    reply_examples = [
        "1 2.10 winner",
        "1 2.00 firsttower",
        "Over 29.5 1.84 kills",
        "kills",
        "28.5 1.68 2.00",
        "29.5 1.84 1.86",
        "30.5 2.00 1.70",
    ]
    if league != "LPL":
        reply_examples.insert(1, "2 1.85 firstblood")
    return (
        "ALERTA MAPA 1\n"
        f"Partido: {team_a} vs {team_b}\n"
        f"Liga: {league}\n"
        f"{scheduled_at_text}"
        f"Equipo 1: {team_a}\n"
        f"Equipo 2: {team_b}\n"
        "Cuotas justas modelo\n"
        f"Winner: Equipo 1 @{float(match_row['winner_team_a_fair_odd']):.2f} | "
        f"Equipo 2 @{float(match_row['winner_team_b_fair_odd']):.2f}\n"
        f"{firstblood_line}"
        f"First Tower: Equipo 1 @{float(match_row['firsttower_team_a_fair_odd']):.2f} | "
        f"Equipo 2 @{float(match_row['firsttower_team_b_fair_odd']):.2f}\n"
        f"Total Kills: {float(match_row['kills_prediction']):.1f}\n"
        "Responde con:\n"
        + "\n".join(reply_examples)
    )


def parse_match_name_from_alert(text: str) -> str | None:
    match = re.search(r"Partido:\s*(.+)$", text, flags=re.MULTILINE)
    return match.group(1).strip() if match else None


def parse_league_from_alert(text: str) -> str | None:
    match = re.search(r"Liga:\s*(.+)$", text, flags=re.MULTILINE)

def parse_team_slots_from_alert(text: str) -> dict[str, str]:
    matches = re.findall(r"Equipo\s+([12]):\s*(.+)$", text, flags=re.MULTILINE)

def load_prediction_row(config: BotConfig, match_name: str) -> pd.Series:
    if not config.predictions_output_path.exists():
        raise LoLPipelineError(
            f"No existe el snapshot de predicciones: {config.predictions_output_path}"
        )
    prediction_frame = pd.read_csv(config.predictions_output_path)
    matches = prediction_frame.loc[prediction_frame["match_name"].astype(str) == match_name]
    if matches.empty:
        raise LoLPipelineError(
            f"No encontre el partido `{match_name}` en el ultimo snapshot de predicciones."
        )
    return matches.iloc[0]


def parse_pick_message(
    text: str,
) -> dict[str, Any] | None:
    normalized = text.strip().replace(",", ".").replace("@", "")
    kills_match = re.fullmatch(
        r"(Over|Under)\s+(\d+(?:\.\d+)?)\s+(\d+(?:\.\d+)?)\s+(kills)",
        normalized,
        flags=re.IGNORECASE,
    )
    if kills_match:
        odd = float(kills_match.group(3))
        if odd <= 1.0:
            return None
        return {
            "market": "kills",
            "selection": kills_match.group(1).title(),
            "line": float(kills_match.group(2)),
            "my_odd": odd,
        }

    binary_match = re.fullmatch(
        r"(.+?)\s+(\d+(?:\.\d+)?)\s+(winner|firstblood|firsttower)",
        normalized,
        flags=re.IGNORECASE,
    )
    if not binary_match:
        return None
    odd = float(binary_match.group(2))
    if odd <= 1.0:
        return None
    return {
        "market": binary_match.group(3).lower(),
        "selection": binary_match.group(1).strip(),
        "line": None,
        "my_odd": odd,
    }


def parse_kills_batch_message(text: str) -> list[dict[str, float]] | None:
    lines = [line.strip() for line in text.strip().splitlines() if line.strip()]
    if not lines:
        return None
    if lines[0].lower() != "kills":
        return None

    options: list[dict[str, float]] = []
    for line in lines[1:]:
        match = re.fullmatch(
            r"(\d+(?:\.\d+)?)\s+(\d+(?:\.\d+)?)\s+(\d+(?:\.\d+)?)",
            line.replace(",", ".").replace("@", "").strip(),
        )
        if not match:
            raise LoLPipelineError(
                "Formato kills invalido. Usa `kills` y luego lineas tipo `29.5 1.84 1.86`."
            )
        options.append(
            {
                "line": float(match.group(1)),
                "over_odd": float(match.group(2)),
                "under_odd": float(match.group(3)),
            }
        )
    if not options:
        return None
    return options


def build_pick_payload_from_reply(
    config: BotConfig,
    reply_text: str,
    alert_text: str,
) -> dict[str, Any]:
    parsed_pick = parse_pick_message(reply_text)
    if parsed_pick is None:
        raise LoLPipelineError(
            "Formato invalido. Usa `Equipo cuota winner`, `Equipo cuota firstblood`, `Equipo cuota firsttower` o `Over linea cuota kills`."
        )

    match_name = parse_match_name_from_alert(alert_text)
    if not match_name:
        raise LoLPipelineError("No pude extraer el partido del mensaje original.")
    league = parse_league_from_alert(alert_text)
    market = parsed_pick["market"]
    my_odd = float(parsed_pick["my_odd"])

    if market not in SUPPORTED_MARKETS:
        raise LoLPipelineError(f"Mercado no soportado: {market}")
    if league == "LPL" and market == "firstblood":
        raise LoLPipelineError("First Blood queda desactivado para LPL en esta version.")

    model_probability: float
    model_fair_odd: float | None
    predicted_value: float | None = None

    if market == "kills":
        predicted_total = parse_kills_prediction_from_alert(alert_text)
        if predicted_total is None:
            raise LoLPipelineError("No pude extraer la prediccion de kills del mensaje original.")
        prediction_row = load_prediction_row(config, match_name)
        residual_std = float(prediction_row.get("kills_residual_std", 6.0))
        line = float(parsed_pick["line"])
        selection = str(parsed_pick["selection"])
        model_probability = estimate_over_under_probability(
            predicted_total=predicted_total,
            line=line,
            residual_std=residual_std,
            direction=selection,
        )
        model_fair_odd = calculate_fair_odd(model_probability)
        predicted_value = predicted_total
    else:
        prediction_row = load_prediction_row(config, match_name)
        team_slots = parse_team_slots_from_alert(alert_text)
        team_map = {
            str(prediction_row["team_a"]): {
                "probability": float(prediction_row[f"{market}_team_a_prob"]),
                "fair_odd": float(prediction_row[f"{market}_team_a_fair_odd"]),
            },
            str(prediction_row["team_b"]): {
                "probability": float(prediction_row[f"{market}_team_b_prob"]),
                "fair_odd": float(prediction_row[f"{market}_team_b_fair_odd"]),
            },
        }
        raw_selection = str(parsed_pick["selection"]).strip()
        slot_aliases = {
            "1": team_slots.get("1", str(prediction_row["team_a"])),
            "2": team_slots.get("2", str(prediction_row["team_b"])),
            "equipo 1": team_slots.get("1", str(prediction_row["team_a"])),
            "equipo1": team_slots.get("1", str(prediction_row["team_a"])),
            "team 1": team_slots.get("1", str(prediction_row["team_a"])),
            "team1": team_slots.get("1", str(prediction_row["team_a"])),
            "equipo 2": team_slots.get("2", str(prediction_row["team_b"])),
            "equipo2": team_slots.get("2", str(prediction_row["team_b"])),
            "team 2": team_slots.get("2", str(prediction_row["team_b"])),
            "team2": team_slots.get("2", str(prediction_row["team_b"])),
        }
        canonical_selection = slot_aliases.get(raw_selection.lower(), raw_selection)
        selection = next(
            (team for team in team_map if team.lower() == canonical_selection.lower()),
            None,
        )
        if selection is None:
            available = ", ".join(team_map.keys())
            raise LoLPipelineError(
                f"No pude identificar la seleccion. Equipos validos: {available}"
            )
        model_probability = float(team_map[selection]["probability"])
        model_fair_odd = float(team_map[selection]["fair_odd"])
        predicted_value = None

    implied_probability = calculate_implied_probability(my_odd)
    edge = model_probability - implied_probability
    ev = calculate_ev(model_probability, my_odd)
    full_kelly = calculate_kelly_fraction(model_probability, my_odd)
    applied_kelly = full_kelly * config.kelly_fraction
    recommended_stake = min(
        config.bankroll_amount * applied_kelly,
        config.max_stake_amount,
    )

    return {
        "match_name": match_name,
        "league": league,
        "market": market,
        "selection": selection,
        "line": parsed_pick["line"],
        "my_odd": my_odd,
        "model_probability": model_probability,
        "model_fair_odd": model_fair_odd,
        "predicted_value": predicted_value,
        "implied_probability": implied_probability,
        "edge": edge,
        "ev": ev,
        "full_kelly": full_kelly,
        "kelly_fraction": config.kelly_fraction,
        "recommended_stake": recommended_stake,
        "bankroll_snapshot": config.bankroll_amount,
        "status": "pending",
    }


def evaluate_kills_batch_from_reply(
    config: BotConfig,
    reply_text: str,
    alert_text: str,
) -> list[dict[str, Any]] | None:
    options = parse_kills_batch_message(reply_text)
    if options is None:
        return None

    match_name = parse_match_name_from_alert(alert_text)
    if not match_name:
        raise LoLPipelineError("No pude extraer el partido del mensaje original.")
    league = parse_league_from_alert(alert_text)
    predicted_total = parse_kills_prediction_from_alert(alert_text)
    if predicted_total is None:
        raise LoLPipelineError("No pude extraer la prediccion de kills del mensaje original.")
    prediction_row = load_prediction_row(config, match_name)
    residual_std = float(prediction_row.get("kills_residual_std", 6.0))

    evaluations: list[dict[str, Any]] = []
    for option in options:
        for direction, odd in [("Over", option["over_odd"]), ("Under", option["under_odd"])]:
            probability = estimate_over_under_probability(
                predicted_total=predicted_total,
                line=float(option["line"]),
                residual_std=residual_std,
                direction=direction,
            )
            fair_odd = calculate_fair_odd(probability)
            implied_probability = calculate_implied_probability(float(odd))
            edge = probability - implied_probability
            ev = calculate_ev(probability, float(odd))
            full_kelly = calculate_kelly_fraction(probability, float(odd))
            applied_kelly = full_kelly * config.kelly_fraction
            recommended_stake = min(
                config.bankroll_amount * applied_kelly,
                config.max_stake_amount,
            )
            evaluations.append(
                {
                    "match_name": match_name,
                    "league": league,
                    "market": "kills",
                    "selection": direction,
                    "line": float(option["line"]),
                    "my_odd": float(odd),
                    "model_probability": probability,
                    "model_fair_odd": fair_odd,
                    "predicted_value": predicted_total,
                    "implied_probability": implied_probability,
                    "edge": edge,
                    "ev": ev,
                    "full_kelly": full_kelly,
                    "kelly_fraction": config.kelly_fraction,
                    "recommended_stake": recommended_stake,
                    "bankroll_snapshot": config.bankroll_amount,
                    "status": "pending",
                }
            )

    evaluations.sort(
        key=lambda item: (
            item["recommended_stake"] > 0,
            item["ev"],
            item["edge"],
        ),
        reverse=True,
    )
    return evaluations

def resolve_column(frame: pd.DataFrame, candidates: list[str], label: str) -> str:
    for candidate in candidates:

    raise LoLPipelineError(

def validate_team_columns(frame: pd.DataFrame) -> tuple[str, str, str]:
    missing_columns = TEAM_REQUIRED_COLUMNS.difference(frame.columns)

    team_column = resolve_column(frame, TEAM_CANDIDATES, "team")

def filter_training_team_rows(raw_frame: pd.DataFrame) -> pd.DataFrame:
    game_id_column, team_column, date_column = validate_team_columns(raw_frame)

    frame = raw_frame.copy()
    frame.columns = [str(column).strip().lower() for column in frame.columns]
    frame["side"] = frame["side"].astype(str).str.strip().str.lower().map(SIDE_MAP)
    frame["league"] = frame["league"].astype(str).str.strip()
    frame["patch"] = frame["patch"].astype(str).str.strip()
    frame["datacompleteness"] = frame["datacompleteness"].astype(str).str.strip().str.lower()
    frame["position"] = frame["position"].astype(str).str.strip().str.lower()
    frame[team_column] = frame[team_column].astype(str).str.strip()
    frame[date_column] = pd.to_datetime(frame[date_column], utc=True, errors="coerce")
    frame["game"] = pd.to_numeric(frame["game"], errors="coerce")
    frame["year"] = pd.to_numeric(frame["year"], errors="coerce")

    for binary_column in ["firstblood", "firsttower", "result"]:
        frame[binary_column] = normalize_binary(frame[binary_column])

    for numeric_column in [
        "firstpick",
        "golddiffat15",
        "xpdiffat15",
        "vspm",
        "ckpm",
        "kills",
        "towers",
        "dragons",
        "barons",
    ]:
        frame[numeric_column] = pd.to_numeric(frame[numeric_column], errors="coerce")

    filtered = frame.loc[
        is_allowed_completeness(frame["league"], frame["datacompleteness"])
        & (frame["position"] == "team")
        & (
            ((frame["year"] == 2025) & (frame["game"] >= 1))
            | ((frame["year"] >= 2026) & (frame["game"] == 1))
        )
        & (frame["league"].isin(LEAGUES_TO_INCLUDE))
        & (frame["side"].isin({"Blue", "Red"}))
    ].copy()

    filtered = filtered.rename(
        columns={
            game_id_column: "game_id",
            team_column: "team_name",
            date_column: "match_date",
        }
    )
    filtered = filtered.dropna(subset=["game_id", "match_date", "team_name"])
    filtered = filtered.sort_values(["team_name", "match_date", "game_id"]).reset_index(drop=True)
    return filtered

def add_multimarket_rolling_features(
    team_frame: pd.DataFrame,
    config: MultiMarketTrainingConfig,
) -> pd.DataFrame:
    frame = team_frame.copy()
    frame["team_games_played_pre"] = frame.groupby("team_name").cumcount()
    frame["team_side_games_played_pre"] = frame.groupby(["team_name", "side"]).cumcount()

    for column in ROLLING_COLUMNS:
        rolling_column = f"{column}_roll{config.rolling_window}"
        frame[rolling_column] = (
            frame.groupby("team_name")[column]
            .transform(
                lambda series: series.shift(1).rolling(
                    window=config.rolling_window,
                    min_periods=1,
                ).mean()
            )
        )
        if column in SIDE_AWARE_COLUMNS:
            side_rolling_column = f"{column}_side_roll{config.rolling_window}"
            frame[side_rolling_column] = (
                frame.groupby(["team_name", "side"])[column]
                .transform(
                    lambda series: series.shift(1).rolling(
                        window=config.rolling_window,
                        min_periods=1,
                    ).mean()
                )
            )

    ordered = frame.sort_values(["match_date", "game_id", "side"]).reset_index(drop=True)
    ordered["league_patch_blue_firstblood_rate_pre"] = (
        ordered.groupby(["league", "patch"])["firstblood"]
        .transform(lambda series: series.shift(1).expanding(min_periods=1).mean())
    )
    ordered["league_patch_blue_win_rate_pre"] = (
        ordered.groupby(["league", "patch"])["result"]
        .transform(lambda series: series.shift(1).expanding(min_periods=1).mean())
    )
    ordered["league_patch_blue_firsttower_rate_pre"] = (
        ordered.groupby(["league", "patch"])["firsttower"]
        .transform(lambda series: series.shift(1).expanding(min_periods=1).mean())
    )
    ordered["league_patch_total_kills_mean_pre"] = (
        ordered.groupby(["league", "patch"])["kills"]
        .transform(lambda series: series.shift(1).expanding(min_periods=1).mean() * 2.0)
    )
    return ordered

def build_classifier_pipeline(
    categorical_features: list[str],
    numeric_features: list[str],
    random_state: int,
) -> Pipeline:
    preprocessor = ColumnTransformer(
        transformers=[
            (
                "categorical",
                Pipeline(
                    steps=[
                        ("imputer", SimpleImputer(strategy="most_frequent")),
                        ("encoder", OneHotEncoder(handle_unknown="ignore")),
                    ]
                ),
                categorical_features,
            ),
            (
                "numeric",
                Pipeline(
                    steps=[
                        ("imputer", SimpleImputer(strategy="median")),
                        ("scaler", StandardScaler()),
                    ]
                ),
                numeric_features,
            ),
        ]
    )
    return Pipeline(
        steps=[
            ("preprocessor", preprocessor),
            ("model", LogisticRegression(max_iter=5000, random_state=random_state)),
        ]
    )

def split_train_test(
    frame: pd.DataFrame,
    config: MultiMarketTrainingConfig,
) -> tuple[pd.DataFrame, pd.DataFrame]:
    if len(frame) < config.min_train_rows:
        raise LoLPipelineError(
            f"Not enough matchup rows to train. Need at least {config.min_train_rows}, got {len(frame)}."
        )
    test_rows = max(1, int(len(frame) * config.test_size))
    train_frame = frame.iloc[:-test_rows].copy()
    test_frame = frame.iloc[-test_rows:].copy()
    if train_frame.empty or test_frame.empty:
        raise LoLPipelineError("Train/test split produced an empty partition.")
    return train_frame, test_frame


def _classification_metrics(y_true: pd.Series, probabilities: pd.Series) -> dict[str, float | None]:
    predictions = (probabilities >= 0.5).astype(int)
    metrics: dict[str, float | None] = {
        "accuracy": float(accuracy_score(y_true, predictions)),
        "log_loss": float(log_loss(y_true, probabilities)),
        "roc_auc": None,
    }
    if y_true.nunique() > 1:
        metrics["roc_auc"] = float(roc_auc_score(y_true, probabilities))
    return metrics


def _regression_metrics(y_true: pd.Series, predictions: pd.Series) -> dict[str, float]:
    rmse = float(math.sqrt(mean_squared_error(y_true, predictions)))
    mae = float(mean_absolute_error(y_true, predictions))
    residual_std = float((y_true - predictions).std(ddof=1)) if len(y_true) > 1 else 1.0
    return {"rmse": rmse, "mae": mae, "residual_std": residual_std}


def train_binary_market_model(
    matchup_frame: pd.DataFrame,
    target_column: str,
    market: str,
    config: MultiMarketTrainingConfig,
) -> dict[str, Any]:
    matchup_frame = matchup_frame.dropna(subset=[target_column]).copy()
    matchup_frame[target_column] = matchup_frame[target_column].astype(int)
    train_frame, test_frame = split_train_test(matchup_frame, config)
    categorical_features, numeric_features, feature_columns = build_market_feature_columns(market)
    candidates: list[dict[str, Any]] = []

    for name, pipeline in [
        (
            "logreg",
            build_classifier_pipeline(categorical_features, numeric_features, config.random_state),
        ),
        (
            "xgboost",
            build_xgb_classifier_pipeline(categorical_features, numeric_features, config.random_state),
        ),
    ]:
        pipeline.fit(train_frame[feature_columns], train_frame[target_column])
        probabilities = pd.Series(
            pipeline.predict_proba(test_frame[feature_columns])[:, 1],
            index=test_frame.index,
        )
        candidates.append(
            {
                "name": name,
                "model": pipeline,
                "metrics": _classification_metrics(test_frame[target_column], probabilities),
            }
        )

    ranked_candidates = sorted(
        candidates,
        key=lambda candidate: (
            candidate["metrics"]["log_loss"],
            -(
                candidate["metrics"]["roc_auc"]
                if candidate["metrics"]["roc_auc"] is not None
                else float("-inf")
            ),
        ),
    )
    champion = ranked_candidates[0]
    return {
        "model": champion["model"],
        "feature_columns": feature_columns,
        "metrics": {
            "train_rows": int(len(train_frame)),
            "test_rows": int(len(test_frame)),
            "selected_model_name": champion["name"],
            "candidate_metrics": {
                candidate["name"]: candidate["metrics"] for candidate in ranked_candidates
            },
            **champion["metrics"],
        },
    }

        candidates,

    return {

def calculate_fair_odd(probability: float) -> float | None:
    if probability <= 0:
        return None
    return 1.0 / probability


def calculate_implied_probability(odd: float) -> float:
    if odd <= 1.0:
        raise LoLPipelineError("Odds must be greater than 1.0.")
    return 1.0 / odd


def calculate_ev(probability: float, odd: float) -> float:
    return probability * odd - 1.0


def calculate_kelly_fraction(probability: float, odd: float) -> float:
    edge = calculate_ev(probability, odd)
    if edge <= 0:
        return 0.0
    b = odd - 1.0
    q = 1.0 - probability
    kelly = ((b * probability) - q) / b
    return max(0.0, float(kelly))


def estimate_over_under_probability(
    predicted_total: float,
    line: float,
    residual_std: float,
    direction: str,
) -> float:
    std = max(float(residual_std), 1.0)
    dist = NormalDist(mu=predicted_total, sigma=std)
    if direction.lower() == "over":
        return float(1.0 - dist.cdf(line))
    if direction.lower() == "under":
        return float(dist.cdf(line))
    raise LoLPipelineError(f"Unsupported direction: {direction}")

def add_rolling_team_features(
    team_frame: pd.DataFrame, config: TrainingConfig
) -> pd.DataFrame:
    frame = team_frame.copy()
    frame["team_games_played_pre"] = frame.groupby("team_name").cumcount()
    frame["team_side_games_played_pre"] = frame.groupby(["team_name", "side"]).cumcount()

    for column in ROLLING_FEATURE_COLUMNS:
        rolling_column = f"{column}_roll{config.rolling_window}"
        frame[rolling_column] = (
            frame.groupby("team_name")[column]
            .transform(
                lambda series: series.shift(1).rolling(
                    window=config.rolling_window,
                    min_periods=1,
                ).mean()
            )
        )
        if column == "firstblood":
            side_rolling_column = f"{column}_side_roll{config.rolling_window}"
            frame[side_rolling_column] = (
                frame.groupby(["team_name", "side"])[column]
                .transform(
                    lambda series: series.shift(1).rolling(
                        window=config.rolling_window,
                        min_periods=1,
                    ).mean()
                )
            )

    frame["firstblood_rate_roll5"] = frame[f"firstblood_roll{config.rolling_window}"]
    frame["firstblood_side_rate_roll5"] = frame[
        f"firstblood_side_roll{config.rolling_window}"
    ]
    frame = frame.drop(columns=[f"firstblood_roll{config.rolling_window}"])
    frame = frame.drop(columns=[f"firstblood_side_roll{config.rolling_window}"])

    ordered_frame = frame.sort_values(["match_date", "game_id", "side"]).reset_index(drop=True)
    ordered_frame["league_patch_blue_firstblood_rate_pre"] = (
        ordered_frame.groupby(["league", "patch"])["firstblood"]
        .transform(lambda series: series.shift(1).expanding(min_periods=1).mean())
    )
    return ordered_frame


def build_matchup_training_frame(team_frame: pd.DataFrame, config: TrainingConfig) -> pd.DataFrame:

            "firstblood_side_rate_roll5": "blue_firstblood_side_rate_roll5",
            "league_patch_blue_firstblood_rate_pre": "league_patch_blue_firstblood_rate_pre",
        }

    red = red[keep_columns].rename(

            "team_name": "red_team",
            "firstpick": "red_has_firstpick",

            "team_games_played_pre": "red_games_played_pre",

    matchup["target_blue_firstblood"] = pd.to_numeric(

        errors="coerce",

        matchup["blue_golddiffat15_roll5"] - matchup["red_golddiffat15_roll5"]

    matchup["diff_ckpm_roll5"] = matchup["blue_ckpm_roll5"] - matchup["red_ckpm_roll5"]
    matchup["diff_firstblood_rate_roll5"] = (
        matchup["blue_firstblood_rate_roll5"] - matchup["red_firstblood_rate_roll5"]
    )
    matchup["diff_firstblood_side_rate_roll5"] = (
        matchup["blue_firstblood_side_rate_roll5"] - matchup["red_firstblood_side_rate_roll5"]
    )

    matchup = matchup.sort_values(["match_date", "game_id"]).reset_index(drop=True)

def build_xgb_pipeline(
    categorical_features: list[str],
    numeric_features: list[str],
    config: TrainingConfig,
) -> Pipeline:
    preprocessor = ColumnTransformer(
        transformers=[
            ("categorical", OneHotEncoder(handle_unknown="ignore"), categorical_features),
            ("numeric", "passthrough", numeric_features),
        ]
    )
    model = XGBClassifier(
        n_estimators=80,
        max_depth=2,
        learning_rate=0.05,
        subsample=0.9,
        colsample_bytree=0.9,
        min_child_weight=5,
        gamma=1.0,
        reg_alpha=0.5,
        reg_lambda=2.0,
        objective="binary:logistic",
        eval_metric="logloss",
        random_state=config.random_state,
    )
    return Pipeline(
        steps=[
            ("preprocessor", preprocessor),
            ("model", model),
        ]
    )


def build_compact_pipeline(
    categorical_features: list[str],
    numeric_features: list[str],
) -> Pipeline:
    compact_preprocessor = ColumnTransformer(
        transformers=[
            (
                "categorical",
                Pipeline(
                    steps=[
                        ("imputer", SimpleImputer(strategy="most_frequent")),
                        ("encoder", OneHotEncoder(handle_unknown="ignore")),
                    ]
                ),
                categorical_features,
            ),
            (
                "numeric",
                Pipeline(
                    steps=[
                        ("imputer", SimpleImputer(strategy="median")),
                        ("scaler", StandardScaler()),
                    ]
                ),
                numeric_features,
            ),
        ]
    )
    return Pipeline(
        steps=[
            ("preprocessor", compact_preprocessor),
            ("model", LogisticRegression(max_iter=5000)),
        ]
    )


def evaluate_candidate_metrics(
    pipeline: Pipeline,
    train_frame: pd.DataFrame,
    test_frame: pd.DataFrame,
    feature_columns: list[str],
) -> dict[str, float | int | None]:
    probabilities = pipeline.predict_proba(test_frame[feature_columns])[:, 1]
    predictions = (probabilities >= 0.5).astype(int)
    metrics: dict[str, float | int | None] = {
        "accuracy": float(accuracy_score(test_frame["target_blue_firstblood"], predictions)),
        "log_loss": float(log_loss(test_frame["target_blue_firstblood"], probabilities)),
        "roc_auc": None,
    }
    if test_frame["target_blue_firstblood"].nunique() > 1:
        metrics["roc_auc"] = float(
            roc_auc_score(test_frame["target_blue_firstblood"], probabilities)
        )
    return metrics


def train_candidate_model(
    name: str,
    pipeline: Pipeline,
    feature_columns: list[str],
    train_frame: pd.DataFrame,
    test_frame: pd.DataFrame,
) -> dict[str, Any]:
    pipeline.fit(train_frame[feature_columns], train_frame["target_blue_firstblood"])
    metrics = evaluate_candidate_metrics(
        pipeline=pipeline,
        train_frame=train_frame,
        test_frame=test_frame,
        feature_columns=feature_columns,
    )
    return {
        "name": name,
        "pipeline": pipeline,
        "feature_columns": feature_columns,
        "metrics": metrics,
    }


def split_train_test(frame: pd.DataFrame, config: TrainingConfig) -> tuple[pd.DataFrame, pd.DataFrame]:
    if len(frame) < config.min_train_rows:
        raise LoLPipelineError(
            f"Not enough matchup rows to train. Need at least {config.min_train_rows}, got {len(frame)}."
        )

    test_rows = max(1, int(len(frame) * config.test_size))
    train_frame = frame.iloc[:-test_rows].copy()
    test_frame = frame.iloc[-test_rows:].copy()
    if train_frame.empty or test_frame.empty:
        raise LoLPipelineError("Train/test split produced an empty partition.")
    return train_frame, test_frame


def train_firstblood_model(
    matchup_frame: pd.DataFrame, config: TrainingConfig
) -> tuple[Pipeline, dict[str, Any], list[str]]:
    matchup_frame = matchup_frame.dropna(subset=["target_blue_firstblood"]).copy()
    matchup_frame["target_blue_firstblood"] = matchup_frame["target_blue_firstblood"].astype(int)
    train_frame, test_frame = split_train_test(matchup_frame, config)
    xgb_categorical_features, xgb_numeric_features = build_feature_columns()
    xgb_feature_columns = xgb_categorical_features + xgb_numeric_features
    compact_categorical_features, compact_numeric_features = build_compact_feature_columns()
    compact_feature_columns = compact_categorical_features + compact_numeric_features

    candidates = [
        train_candidate_model(
            name="xgboost_full",
            pipeline=build_xgb_pipeline(xgb_categorical_features, xgb_numeric_features, config),
            feature_columns=xgb_feature_columns,
            train_frame=train_frame,
            test_frame=test_frame,
        ),
        train_candidate_model(
            name="logreg_compact",
            pipeline=build_compact_pipeline(
                compact_categorical_features,
                compact_numeric_features,
            ),
            feature_columns=compact_feature_columns,
            train_frame=train_frame,
            test_frame=test_frame,
        ),
    ]

    ranked_candidates = sorted(
        candidates,
        key=lambda candidate: (
            candidate["metrics"]["log_loss"],
            -(
                candidate["metrics"]["roc_auc"]
                if candidate["metrics"]["roc_auc"] is not None
                else float("-inf")
            ),
        ),
    )
    champion = ranked_candidates[0]
    metrics: dict[str, Any] = {
        "train_rows": int(len(train_frame)),
        "test_rows": int(len(test_frame)),
        "selected_model_name": str(champion["name"]),
        "candidate_metrics": {
            candidate["name"]: candidate["metrics"] for candidate in ranked_candidates
        },
        "accuracy": champion["metrics"]["accuracy"],
        "log_loss": champion["metrics"]["log_loss"],
        "roc_auc": champion["metrics"]["roc_auc"],
    }
    return champion["pipeline"], metrics, champion["feature_columns"]

def fetch_table(config: DashboardConfig, table_name: str) -> pd.DataFrame:
    url = f"{config.supabase_url}/rest/v1/{table_name}"
    params = {"select": "*", "order": "created_at.desc", "limit": "5000"}
    response = requests.get(url, headers=_supabase_headers(config), params=params, timeout=30)
    if response.status_code == 404:
        return pd.DataFrame()
    if response.status_code >= 400:
        raise LoLPipelineError(
            f"Supabase fetch failed for {table_name} ({response.status_code}): {response.text}"
        )
    payload = response.json()
    return pd.DataFrame(payload)


def update_table_row(
    config: DashboardConfig,
    table_name: str,
    row_id: str,
    payload: dict[str, Any],
) -> None:
    url = f"{config.supabase_url}/rest/v1/{table_name}"
    headers = {
        "apikey": config.supabase_service_role_key,
        "Authorization": f"Bearer {config.supabase_service_role_key}",
        "Content-Type": "application/json",
        "Accept-Profile": config.supabase_schema,
        "Content-Profile": config.supabase_schema,
        "Prefer": "return=minimal",
    }
    response = requests.patch(
        url,
        headers=headers,
        params={"id": f"eq.{row_id}"},
        json=payload,
        timeout=30,
    )
    if response.status_code >= 400:
        raise LoLPipelineError(
            f"Supabase update failed for {table_name}/{row_id} ({response.status_code}): {response.text}"
        )

        download_kwargs["url"] = drive_reference

    downloaded = gdown.download(
        **download_kwargs,

    if (not downloaded or not output_path.exists()) and file_id:

                    "datacompleteness",

def settle_legacy_bets(frame: pd.DataFrame, results_frame: pd.DataFrame) -> pd.DataFrame:
    if frame.empty:
        return frame

    settled = frame.copy()
    settled["match_name"] = settled["match_name"].astype(str)
    settled["created_at"] = pd.to_datetime(settled["created_at"], utc=True, errors="coerce")
    settled["picked_team"] = settled.get("picked_team").fillna(settled.get("model_selected_team"))
    settled["my_odd"] = pd.to_numeric(settled["my_odd"], errors="coerce")
    settled["model_fair_odd"] = pd.to_numeric(settled["model_fair_odd"], errors="coerce")
    settled = settled.loc[settled["my_odd"] > settled["model_fair_odd"]].copy()
    if results_frame.empty:
        settled["settled_status"] = settled.get("status", "pending")
        settled["settled_league"] = pd.NA
        settled["settled_match_date"] = pd.NaT
        return settled

    status_values: list[str] = []
    settled_leagues: list[str | None] = []
    settled_dates: list[pd.Timestamp | None] = []
    for _, row in settled.iterrows():
        team_parts = [part.strip() for part in str(row["match_name"]).split(" vs ")]
        if len(team_parts) != 2:
            status_values.append(str(row.get("status", "pending")).lower())
            settled_leagues.append(None)
            settled_dates.append(None)
            continue
        pair_key = "||".join(sorted([normalize_name_key(team_parts[0]), normalize_name_key(team_parts[1])]))
        created_at = row["created_at"]
        candidates = results_frame.loc[results_frame["pair_key"] == pair_key].copy()
        if pd.notna(created_at):
            candidates = candidates.loc[candidates["match_date"] >= created_at]
        candidates = candidates.sort_values("match_date")
        if candidates.empty:
            status_values.append(str(row.get("status", "pending")).lower())
            settled_leagues.append(None)
            settled_dates.append(None)
            continue
        match_row = candidates.iloc[0]
        winner = match_row["firstblood_winner"]
        picked_team = str(row["picked_team"]).strip()
        if not winner or pd.isna(winner):
            status_values.append("void")
        elif normalize_name_key(winner) == normalize_name_key(picked_team):
            status_values.append("won")
        else:
            status_values.append("lost")
        settled_leagues.append(str(match_row["league"]))
        settled_dates.append(match_row["match_date"])

    settled["settled_status"] = status_values
    settled["settled_league"] = settled_leagues
    settled["settled_match_date"] = pd.to_datetime(settled_dates, utc=True, errors="coerce")
    return settled


def _find_settlement_row(
    match_name: str,
    created_at: Any,
    results_frame: pd.DataFrame,
) -> pd.Series | None:
    team_parts = [part.strip() for part in str(match_name).split(" vs ")]
    if len(team_parts) != 2:
        return None
    pair_key = "||".join(sorted([normalize_name_key(team_parts[0]), normalize_name_key(team_parts[1])]))
    candidates = results_frame.loc[results_frame["pair_key"] == pair_key].copy()
    created_at_ts = pd.to_datetime(created_at, utc=True, errors="coerce")
    if pd.notna(created_at_ts):
        candidates = candidates.loc[candidates["match_date"] >= created_at_ts]
    candidates = candidates.sort_values("match_date")
    if candidates.empty:
        return None
    return candidates.iloc[0]


def settle_picks(frame: pd.DataFrame, results_frame: pd.DataFrame) -> pd.DataFrame:
    if frame.empty:
        return frame

    settled = frame.copy()
    if results_frame.empty:
        settled["settled_status"] = settled.get("status", "pending")
        settled["settled_league"] = settled.get("league")
        settled["settled_match_date"] = pd.NaT
        settled["settled_actual_value"] = pd.NA
        return settled

    settled["created_at"] = pd.to_datetime(settled["created_at"], utc=True, errors="coerce")
    settled["line"] = pd.to_numeric(settled.get("line"), errors="coerce")
    settled["predicted_value"] = pd.to_numeric(settled.get("predicted_value"), errors="coerce")

    status_values: list[str] = []
    settled_leagues: list[str | None] = []
    settled_dates: list[pd.Timestamp | None] = []
    settled_actual_values: list[float | str | None] = []

    for _, row in settled.iterrows():
        match_row = _find_settlement_row(str(row.get("match_name", "")), row.get("created_at"), results_frame)
        if match_row is None:
            status_values.append(str(row.get("status", "pending")).lower())
            settled_leagues.append(row.get("league"))
            settled_dates.append(None)
            settled_actual_values.append(None)
            continue

        market = str(row.get("market", "")).strip().lower()
        selection = str(row.get("selection", "")).strip()
        status = str(row.get("status", "pending")).lower()
        actual_value: float | str | None = None

        if market == "winner":
            actual_value = match_row.get("winner")
            if not actual_value or pd.isna(actual_value):
                status = "void"
            elif normalize_name_key(str(actual_value)) == normalize_name_key(selection):
                status = "won"
            else:
                status = "lost"
        elif market == "firstblood":
            actual_value = match_row.get("firstblood_winner")
            if not actual_value or pd.isna(actual_value):
                status = "void"
            elif normalize_name_key(str(actual_value)) == normalize_name_key(selection):
                status = "won"
            else:
                status = "lost"
        elif market == "firsttower":
            actual_value = match_row.get("firsttower_winner")
            if not actual_value or pd.isna(actual_value):
                status = "void"
            elif normalize_name_key(str(actual_value)) == normalize_name_key(selection):
                status = "won"
            else:
                status = "lost"
        elif market == "kills":
            actual_total = pd.to_numeric(match_row.get("total_kills"), errors="coerce")
            line_value = pd.to_numeric(row.get("line"), errors="coerce")
            actual_value = None if pd.isna(actual_total) else float(actual_total)
            if pd.isna(actual_total) or pd.isna(line_value):
                status = "pending"
            elif selection.lower() == "over":
                status = "won" if float(actual_total) > float(line_value) else "lost" if float(actual_total) < float(line_value) else "void"
            elif selection.lower() == "under":
                status = "won" if float(actual_total) < float(line_value) else "lost" if float(actual_total) > float(line_value) else "void"
            else:
                status = "pending"

        status_values.append(status)
        settled_leagues.append(str(match_row.get("league")) if pd.notna(match_row.get("league")) else row.get("league"))
        settled_dates.append(match_row.get("match_date"))
        settled_actual_values.append(actual_value)

    settled["settled_status"] = status_values
    settled["settled_league"] = settled_leagues
    settled["settled_match_date"] = pd.to_datetime(settled_dates, utc=True, errors="coerce")
    settled["settled_actual_value"] = settled_actual_values
    return settled

def calculate_pick_pnl(row: pd.Series) -> float | None:
    stake = row.get("recommended_stake")
    odd = row.get("my_odd")
    status = str(row.get("status", "")).lower()
    if pd.isna(stake) or pd.isna(odd):
        return None
    if status == "won":
        return float(stake) * (float(odd) - 1.0)
    if status == "lost":
        return -float(stake)
    if status == "void":
        return 0.0
    return None


def build_unified_records(
    bets_frame: pd.DataFrame,
    picks_frame: pd.DataFrame,
    results_frame: pd.DataFrame,
) -> pd.DataFrame:
    unified = pd.concat(
        [normalize_legacy_bets(bets_frame, results_frame), normalize_settled_picks(picks_frame, results_frame)],
        ignore_index=True,
    )
    if unified.empty:
        return unified
    unified["market"] = unified["market"].astype(str).str.strip()
    unified["league"] = unified["league"].astype(str).replace({"nan": pd.NA})
    unified["status"] = unified["status"].astype(str).str.strip().str.lower()
    unified = unified.sort_values("created_at", ascending=False, na_position="last").reset_index(drop=True)
    return unified


def compute_summary_metrics(records: pd.DataFrame) -> dict[str, Any]:
    if records.empty:
        return {
            "total_records": 0,
            "pending_records": 0,
            "settled_records": 0,
            "won_records": 0,
            "lost_records": 0,
            "void_records": 0,
            "total_stake": 0.0,
            "total_pnl": 0.0,
            "roi": None,
            "avg_edge": None,
            "avg_ev": None,
            "current_bankroll": None,
        }

    pending_mask = records["status"].eq("pending")
    settled_mask = records["status"].isin(["won", "lost", "void"])
    total_stake = float(records.loc[settled_mask, "recommended_stake"].fillna(0.0).sum())
    total_pnl = float(records.loc[settled_mask, "pnl"].fillna(0.0).sum())

    return {
        "total_records": int(len(records)),
        "pending_records": int(pending_mask.sum()),
        "settled_records": int(settled_mask.sum()),
        "won_records": int(records["status"].eq("won").sum()),
        "lost_records": int(records["status"].eq("lost").sum()),
        "void_records": int(records["status"].eq("void").sum()),
        "total_stake": total_stake,
        "total_pnl": total_pnl,
        "roi": (total_pnl / total_stake) if total_stake > 0 else None,
        "avg_edge": (
            float(records["edge"].dropna().mean()) if records["edge"].notna().any() else None
        ),
        "avg_ev": float(records["ev"].dropna().mean()) if records["ev"].notna().any() else None,
        "current_bankroll": None,
    }


def compute_current_bankroll(records: pd.DataFrame, initial_bankroll: float) -> float:
    if records.empty:
        return float(initial_bankroll)
    settled_mask = records["status"].isin(["won", "lost", "void"])
    settled_pnl = float(records.loc[settled_mask, "pnl"].fillna(0.0).sum())
    return float(initial_bankroll) + settled_pnl


def apply_filters(
    records: pd.DataFrame,
    source_filter: list[str],
    league_filter: list[str],
    market_filter: list[str],
    status_filter: list[str],
) -> pd.DataFrame:

        filtered = filtered.loc[filtered["market"].isin(market_filter)]
    if status_filter:

    return filtered.reset_index(drop=True)

def _preprocess_for_ocr(image_bytes: bytes) -> Image.Image:
    image_array = np.frombuffer(image_bytes, dtype=np.uint8)
    image = cv2.imdecode(image_array, cv2.IMREAD_COLOR)
    if image is None:
        raise ValueError("No pude decodificar la imagen para OCR.")

    gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
    gray = cv2.resize(gray, None, fx=2, fy=2, interpolation=cv2.INTER_CUBIC)
    gray = cv2.convertScaleAbs(gray, alpha=1.5, beta=0)

    _, binary = cv2.threshold(gray, 160, 255, cv2.THRESH_BINARY)
    if np.mean(binary) < 127:
        binary = cv2.bitwise_not(binary)

    return Image.fromarray(binary)

def fuzzy_match_team(
    raw_name: str,
    valid_teams_list: list[str],
    confidence_threshold: int = 60,
) -> tuple[str | None, int]:
    cleaned_raw_name = _normalize_for_fuzzy(raw_name)
    if not valid_teams_list:
        return None, 0
    if not cleaned_raw_name:
        return str(valid_teams_list[0]), 0

    normalized_team_map = {
        _normalize_for_fuzzy(team_name): str(team_name)
        for team_name in valid_teams_list
        if _normalize_for_fuzzy(team_name)
    }
    if not normalized_team_map:
        return None, 0

    try:
        scored_matches = [
            (team_name, fuzz.partial_ratio(normalized_team_name, cleaned_raw_name))
            for normalized_team_name, team_name in normalized_team_map.items()
        ]
    except Exception:
        return None, 0

    if not scored_matches:
        return str(valid_teams_list[0]), 0

    team_name, score = max(scored_matches, key=lambda item: int(item[1]))
    score = int(score)
    if score < 30:
        return "No identificado", score
    return str(team_name), score


def _score_team_in_line(team_name: str, normalized_line: str) -> int:
    normalized_team = _normalize_for_keyword_scan(team_name)
    if not normalized_team or not normalized_line:
        return 0

    scores = [fuzz.partial_ratio(normalized_team, normalized_line)]
    for token in normalized_team.replace(".", " ").split():
        if len(token) >= 3 and token not in GENERIC_TEAM_TOKENS:
            scores.append(fuzz.partial_ratio(token, normalized_line))
    return int(max(scores))

    valid_teams_list: list[str],
    minimum_score: int = 70,

    for raw_line in raw_lines:
        cleaned_line = apply_ocr_team_corrections(raw_line)

            score = _score_team_in_line(team_name, normalized_line)

                hits.append({"team": str(team_name), "score": int(score)})

        hits = sorted(hits, key=lambda hit: int(hit["score"]), reverse=True)
        unique_hits: list[dict[str, object]] = []

            continue

        team_a = unique_hits[0]
        team_b = unique_hits[1] if len(unique_hits) > 1 else {"team": "", "score": 0}
        dedupe_pair = tuple(sorted([str(team_a["team"]), str(team_b["team"])]))
        seen_key = (str(dedupe_pair[0]), str(dedupe_pair[1]), cleaned_line)
        if seen_key in seen_keys:
            continue
        seen_keys.add(seen_key)

        results.append(
            {
                "raw_line": cleaned_line,
                "team_a": team_a["team"],
                "score_a": team_a["score"],
                "team_b": team_b["team"],
                "score_b": team_b["score"],
            }
        )

    return results

class SupabaseResponse:
    def __init__(self, data: list[dict[str, Any]] | dict[str, Any] | None) -> None:

class SupabaseQuery:
    def __init__(self, base_url: str, service_role_key: str, schema_name: str, table_name: str) -> None:
        self.base_url = base_url.rstrip("/")
        self.service_role_key = service_role_key
        self.schema_name = schema_name
        self.table_name = table_name
        self.method = "GET"
        self.params: dict[str, str] = {}
        self.payload: dict[str, Any] | None = None

    def select(self, columns: str) -> "SupabaseQuery":
        self.method = "GET"
        self.params["select"] = columns
        return self

    def eq(self, column: str, value: Any) -> "SupabaseQuery":
        self.params[column] = f"eq.{value}"
        return self

    def order(self, column: str, desc: bool = False) -> "SupabaseQuery":
        direction = "desc" if desc else "asc"
        self.params["order"] = f"{column}.{direction}"
        return self

    def limit(self, count: int) -> "SupabaseQuery":
        self.params["limit"] = str(count)
        return self

    def insert(self, data: dict[str, Any]) -> "SupabaseQuery":
        self.method = "POST"
        self.payload = data
        return self

    def execute(self) -> SupabaseResponse:
        headers = {
            "apikey": self.service_role_key,
            "Authorization": f"Bearer {self.service_role_key}",
            "Accept-Profile": self.schema_name,
            "Content-Profile": self.schema_name,
        }
        url = f"{self.base_url}/rest/v1/{self.table_name}"
        if self.method == "POST":
            headers["Content-Type"] = "application/json"
            headers["Prefer"] = "return=representation"
            response = requests.post(url, headers=headers, params=self.params, json=self.payload, timeout=30)
        else:
            response = requests.get(url, headers=headers, params=self.params, timeout=30)

        if response.status_code >= 400:
            raise TennisEVCalculatorError(
                f"Supabase {self.method} failed for {self.schema_name}.{self.table_name} "
                f"({response.status_code}): {response.text}"
            )
        if not response.text:
            return SupabaseResponse([])
        return SupabaseResponse(response.json())

def normalize_name(name: object) -> str:
    text = "" if pd.isna(name) else str(name)
    text = unicodedata.normalize("NFKD", text)
    text = "".join(char for char in text if not unicodedata.combining(char))
    text = text.lower().replace(".", " ")
    text = re.sub(r"[^a-z0-9]+", " ", text).strip()
    return re.sub(r"\s+", " ", text)


def name_score(api_name: object, sackmann_name: object) -> float:
    api = normalize_name(api_name)
    sackmann = normalize_name(sackmann_name)
    if not api or not sackmann:
        return 0.0
    if api == sackmann:
        return 1.0

    api_tokens = api.split()
    sackmann_tokens = sackmann.split()
    api_set = set(api_tokens)
    sackmann_set = set(sackmann_tokens)
    token_overlap = len(api_set & sackmann_set) / max(len(api_set | sackmann_set), 1)

    api_last = api_tokens[-1]
    sackmann_last = sackmann_tokens[-1]
    last_name_score = 0.86 if api_last == sackmann_last else 0.0

    initial_score = 0.0
    if len(api_tokens) == 2 and len(sackmann_tokens) >= 2:
        if api_tokens[0][:1] == sackmann_tokens[0][:1] and api_tokens[-1] == sackmann_tokens[-1]:
            initial_score = 0.95
        if api_tokens[-1][:1] == sackmann_tokens[0][:1] and api_tokens[0] == sackmann_tokens[-1]:
            initial_score = 0.95

    return max(token_overlap, last_name_score, initial_score)

def probability_from_distribution(distribution: dict[int, int], predicate: Any) -> float:
    total = sum(distribution.values())
    if total <= 0:
        return 0.0
    wins = sum(count for value, count in distribution.items() if predicate(int(value)))
    return wins / total


def calculate_ev(probability: float, decimal_odds: float) -> float:
    return probability * decimal_odds - 1.0


def insert_value_bets_to_supabase(value_bets: pd.DataFrame) -> int:
    if value_bets.empty or "EV" not in value_bets.columns:
        return 0

    positives = value_bets.loc[pd.to_numeric(value_bets["EV"], errors="coerce") > 0].copy()
    if positives.empty:
        return 0

    supabase = get_supabase_client()
    inserted = 0
    for _, row in positives.iterrows():
        partido = str(row["partido"])
        mercado = str(row["mercado"])
        linea = float(row["linea"])
        casa_apuestas = str(row["casa_apuestas"])
        cuota_actual = float(row["cuota_casa"])
        ev_actual = float(row["EV"])

            .select("id")
            .eq("partido", partido)
            .eq("mercado", mercado)
            .eq("linea", linea)

        )

        if len(response.data or []) == 0:

                "partido": partido,
                "mercado": mercado,
                "linea": linea,
                "casa_apuestas": casa_apuestas,
                "cuota_inicial": cuota_actual,
                "nuestra_prob": float(row["nuestra_prob"]),
                "ev_inicial": ev_actual,
            }
            insert_response = (
                supabase.schema("tennis")
                .table(TENNIS_PICKS_TABLE)
                .insert(payload)
                .execute()
            )
            inserted_rows = insert_response.data or []
            if not inserted_rows or "id" not in inserted_rows[0]:
                raise TennisEVCalculatorError(f"Supabase did not return id for inserted pick: {partido}")
            pick_id = inserted_rows[0]["id"]
            print(f"[Supabase] Pick insertado: {mercado} {linea} {partido} ({casa_apuestas})")
            inserted += 1
        else:
            pick_id = response.data[0]["id"]

        latest_history = (

            .select("cuota")
            .eq("pick_id", pick_id)

            .limit(1)

        )
        latest_rows = latest_history.data or []

def evaluate_event(
    event_odds: pd.DataFrame,
    player_a_state: pd.Series,
    player_b_state: pd.Series,
    player_a_name: str,
    player_b_name: str,
    best_of: int,
    n_simulations: int,
) -> list[dict[str, Any]]:
    simulation_row = build_simulation_row(player_a_state, player_b_state)
    p_serve_a, p_serve_b = calculate_match_serve_probs(simulation_row)
    simulation = run_monte_carlo(p_serve_a, p_serve_b, best_of=best_of, n_simulations=n_simulations)
    total_distribution = simulation["total_games_distribution"]
    diff_distribution = simulation["game_diff_distribution"]
    rows: list[dict[str, Any]] = []

    for _, odds_row in event_odds.iterrows():
        market = str(odds_row["market"])
        selection = str(odds_row["selection"])
        line = odds_row["line"]
        odds = float(odds_row["odds"])
        probability: float | None = None
        market_label: str | None = None

        if market == "totals" and pd.notna(line):
            line_value = float(line)
            if selection.lower() == "over":
                probability = probability_from_distribution(total_distribution, lambda total: total > line_value)
                market_label = "Over"
            elif selection.lower() == "under":
                probability = probability_from_distribution(total_distribution, lambda total: total < line_value)
                market_label = "Under"

        elif market == "spreads" and pd.notna(line):
            line_value = float(line)
            if name_score(selection, player_a_name) >= name_score(selection, player_b_name):
                probability = probability_from_distribution(diff_distribution, lambda diff: diff > -line_value)
                market_label = f"Handicap {player_a_name}"
            else:
                probability = probability_from_distribution(diff_distribution, lambda diff: diff < line_value)
                market_label = f"Handicap {player_b_name}"

        if probability is None or market_label is None:
            continue

        rows.append(
            {
                "partido": f"{player_a_name} vs {player_b_name}",
                "mercado": market_label,
                "linea": float(line),
                "casa_apuestas": odds_row.get("casa_apuestas") or odds_row.get("bookmaker_title"),
                "cuota_casa": odds,
                "nuestra_prob": probability,
                "EV": calculate_ev(probability, odds),
                "p_serve_A": p_serve_a,
                "p_serve_B": p_serve_b,
                "mean_total_games": simulation["mean_total_games"],
                "mean_game_diff": simulation["mean_game_diff"],
            }
        )

    return rows


def calculate_value_bets(
    features_path: Path = FEATURES_PATH,
    output_path: Path = OUTPUT_PATH,
    best_of: int = 3,
    n_simulations: int = 1000,
) -> pd.DataFrame:
    if not features_path.exists():
        raise TennisEVCalculatorError(f"Features file not found: {features_path}")

    random.seed(42)
    odds = fetch_and_save_odds(ODDS_PATH)
    features = pd.read_csv(features_path)
    latest_state = build_latest_player_state(features)

    output_rows: list[dict[str, Any]] = []
    event_columns = ["event_id", "jugador_a_api", "jugador_b_api"]
    for _, event in odds[event_columns].drop_duplicates().iterrows():
        player_a_state, score_a = find_player_state(str(event["jugador_a_api"]), latest_state)
        player_b_state, score_b = find_player_state(str(event["jugador_b_api"]), latest_state)
        if player_a_state is None or player_b_state is None:
            print(
                "Skipping event due to low name match:",
                event["jugador_a_api"],
                score_a,
                event["jugador_b_api"],
                score_b,
            )
            continue

        event_odds = odds.loc[odds["event_id"].eq(event["event_id"])]
        output_rows.extend(
            evaluate_event(
                event_odds=event_odds,
                player_a_state=player_a_state,
                player_b_state=player_b_state,
                player_a_name=str(player_a_state["player_name"]),
                player_b_name=str(player_b_state["player_name"]),
                best_of=best_of,
                n_simulations=n_simulations,
            )
        )

    output = pd.DataFrame(output_rows)
    if not output.empty:
        output["Kelly Recomendado"] = 1000.0 * output["EV"] / (output["cuota_casa"] - 1.0)
        final_columns = [
            "partido",
            "mercado",
            "linea",
            "casa_apuestas",
            "cuota_casa",
            "nuestra_prob",
            "EV",
            "Kelly Recomendado",
            "p_serve_A",
            "p_serve_B",
            "mean_total_games",
            "mean_game_diff",
        ]
        output = output.sort_values("cuota_casa", ascending=False)

def fetch_odds_for_sport(api_key: str, sport_key: str) -> list[dict[str, Any]]:
    url = f"{THE_ODDS_API_BASE_URL}/sports/{sport_key}/odds"
    params = {

        "markets": MARKETS,

        "dateFormat": "iso",
    }
    response = requests.get(url, params=params, timeout=30)
    if response.status_code >= 400:

def discover_active_tennis_sports(api_key: str) -> list[str]:
    response = requests.get(
        f"{THE_ODDS_API_BASE_URL}/sports",
        params={"apiKey": api_key, "all": "false"},
        timeout=30,
    )
    if response.status_code >= 400:
        raise TennisOddsListenerError(f"The-Odds-API sports error {response.status_code}: {response.text}")
    sports = response.json()
    atp_keys = [
        str(sport["key"])
        for sport in sports
        if str(sport.get("key", "")).startswith("tennis_atp")
    ]
    if atp_keys:
        return sorted(set(atp_keys))

    tennis_keys = [
        str(sport["key"])
        for sport in sports
        if str(sport.get("key", "")).startswith("tennis_")
    ]
    return sorted(set(tennis_keys))


def fetch_market_odds() -> list[dict[str, Any]]:
    api_key = os.getenv("THE_ODDS_API_KEY", "").strip() or os.getenv("ODDS_API_KEY", "").strip()
    if not api_key:
        print("THE_ODDS_API_KEY not found. Using mock tennis odds.")
        return get_mock_odds()
    try:
        return fetch_odds_from_api(api_key)
    except TennisOddsListenerError as exc:
        if "UNKNOWN_SPORT" not in str(exc):
            raise

    active_sports = discover_active_tennis_sports(api_key)
    if not active_sports:
        print("No active tennis sport keys found. Using mock tennis odds.")
        return get_mock_odds()

    events: list[dict[str, Any]] = []
    for sport_key in active_sports:
        try:
            sport_events = fetch_odds_for_sport(api_key, sport_key)
        except TennisOddsListenerError as exc:
            print(f"Skipping {sport_key}: {exc}")
            continue
        events.extend(sport_events)
    if not events:
        print("No tennis odds returned by active sport keys. Using mock tennis odds.")
        return get_mock_odds()
    return events


def normalize_market_odds(events: list[dict[str, Any]]) -> pd.DataFrame:
    rows: list[dict[str, Any]] = []
    for event in events:
        player_a = str(event.get("home_team") or "")
        player_b = str(event.get("away_team") or "")
        for bookmaker in event.get("bookmakers", []) or []:
            bookmaker_key = bookmaker.get("key")
            bookmaker_title = bookmaker.get("title")
            for market in bookmaker.get("markets", []) or []:
                market_key = str(market.get("key", ""))
                if market_key not in {"h2h", "totals", "spreads"}:
                    continue
                for outcome in market.get("outcomes", []) or []:
                    line = outcome.get("point")
                    if market_key == "totals" and line is not None and float(line) < 15:
                        continue
                    rows.append(
                        {
                            "event_id": event.get("id"),
                            "sport_key": event.get("sport_key"),
                            "commence_time": event.get("commence_time"),
                            "jugador_a_api": player_a,
                            "jugador_b_api": player_b,
                            "bookmaker_key": bookmaker_key,
                            "bookmaker_title": bookmaker_title,
                            "casa_apuestas": bookmaker_title,
                            "market": market_key,
                            "selection": outcome.get("name"),
                            "line": line,
                            "odds": float(outcome.get("price")),
                        }
                    )
    return pd.DataFrame(rows)


def build_quote_key(row: pd.Series | dict[str, Any]) -> str:

            str(row.get("bookmaker_key") or ""),
            str(row.get("market") or ""),

            line_key,
        ]
    )


def build_upcoming_odds_rows(frame: pd.DataFrame) -> list[dict[str, Any]]:

def fetch_existing_upcoming_prices() -> dict[str, float]:
    existing = fetch_tennis_table(
        UPCOMING_ODDS_TABLE,
        select="quote_key,price",
        filters={"quote_key": "not.is.null"},
    )
    if existing.empty:
        return {}

    prices: dict[str, float] = {}
    for _, row in existing.iterrows():
        quote_key = row.get("quote_key")
        price = row.get("price")
        if pd.isna(quote_key) or pd.isna(price):
            continue
        prices[str(quote_key)] = float(price)
    return prices


def filter_changed_upcoming_odds_rows(rows: list[dict[str, Any]]) -> list[dict[str, Any]]:
    if not rows:
        return []

    existing_prices = fetch_existing_upcoming_prices()
    changed_rows: list[dict[str, Any]] = []
    for row in rows:
        quote_key = str(row["quote_key"])
        current_price = float(row["price"])
        existing_price = existing_prices.get(quote_key)
        if existing_price is None or abs(existing_price - current_price) >= 1e-9:
            changed_rows.append(row)
    return changed_rows


def save_upcoming_odds_to_supabase(frame: pd.DataFrame) -> int:
    rows = build_upcoming_odds_rows(frame)
    changed_rows = filter_changed_upcoming_odds_rows(rows)
    if not changed_rows:
        return 0

    upsert_tennis_rows(
        UPCOMING_ODDS_TABLE,
        changed_rows,
        on_conflict=UPCOMING_ODDS_CONFLICT_KEY,
    )
    return len(changed_rows)

    output_path.parent.mkdir(parents=True, exist_ok=True)
    frame.to_csv(output_path, index=False)


def run(output_path: Path = OUTPUT_PATH) -> pd.DataFrame:

    except TennisDatabaseError as exc:
        print(f"[Supabase] Upcoming odds not saved: {exc}")
    return frame

    print(odds_frame.to_string(index=False))

from __future__ import annotations

import argparse
from pathlib import Path

import pandas as pd


BASE_DIR = Path(__file__).resolve().parent.parent
RAW_DIR = BASE_DIR / "data" / "raw" / "sackmann"
OUTPUT_PATH = BASE_DIR / "data" / "processed" / "atp_features_avanzadas.csv"
ROLLING_WINDOW = 20

INCOMPLETE_SCORE_TOKENS = {"RET", "W/O", "WO", "DEF", "ABD", "ABN", "Walkover"}


class TennisDataPrepError(RuntimeError):
    """Raised when advanced ATP features cannot be built."""


def parse_set_scores(score: object) -> list[tuple[int, int]]:
    if pd.isna(score):
        return []
    score_text = str(score).strip()
    if not score_text or any(token in score_text for token in INCOMPLETE_SCORE_TOKENS):
        return []

    sets: list[tuple[int, int]] = []
    for raw_set in score_text.split():
        if "[" in raw_set or "]" in raw_set:
            continue
        clean_set = raw_set.split("(", 1)[0]
        if "-" not in clean_set:
            continue
        left, right = clean_set.split("-", 1)
        if left.isdigit() and right.isdigit():
            sets.append((int(left), int(right)))
    return sets


def total_games(score: object) -> int | None:
    sets = parse_set_scores(score)
    if not sets:
        return None
    return sum(winner_games + loser_games for winner_games, loser_games in sets)


def game_difference(score: object) -> int | None:
    sets = parse_set_scores(score)
    if not sets:
        return None
    return sum(winner_games - loser_games for winner_games, loser_games in sets)


def load_raw_matches(raw_dir: Path = RAW_DIR) -> pd.DataFrame:
    paths = sorted(raw_dir.glob("atp_matches_*.csv"))
    if not paths:
        raise TennisDataPrepError(f"No raw Sackmann CSVs found in {raw_dir}")

    frame = pd.concat((pd.read_csv(path) for path in paths), ignore_index=True)
    frame["fecha"] = pd.to_datetime(frame["tourney_date"].astype(str), format="%Y%m%d", errors="coerce")
    frame["source_row_id"] = range(len(frame))
    frame["total_juegos"] = frame["score"].map(total_games)
    frame["diferencia_juegos"] = frame["score"].map(game_difference)
    frame = frame.dropna(subset=["fecha", "winner_id", "loser_id", "score", "total_juegos", "diferencia_juegos"])
    frame["total_juegos"] = frame["total_juegos"].astype(int)
    frame["diferencia_juegos"] = frame["diferencia_juegos"].astype(int)
    return frame.sort_values(["fecha", "tourney_id", "match_num", "source_row_id"]).reset_index(drop=True)


def safe_divide(numerator: pd.Series, denominator: pd.Series) -> pd.Series:
    denominator = denominator.replace(0, pd.NA)
    return numerator / denominator


def build_player_match_rows(matches: pd.DataFrame) -> pd.DataFrame:
    winner_rows = pd.DataFrame(
        {
            "source_row_id": matches["source_row_id"],
            "fecha": matches["fecha"],
            "tourney_id": matches["tourney_id"],
            "match_num": matches["match_num"],
            "player_id": matches["winner_id"],
            "player_name": matches["winner_name"],
            "opponent_id": matches["loser_id"],
            "opponent_name": matches["loser_name"],
            "is_winner": 1,
            "minutes": matches["minutes"],
            "serve_points": matches["w_svpt"],
            "serve_points_won": matches["w_1stWon"] + matches["w_2ndWon"],
            "return_points": matches["l_svpt"],
            "return_points_won": matches["l_svpt"] - matches["l_1stWon"] - matches["l_2ndWon"],
        }
    )
    loser_rows = pd.DataFrame(
        {
            "source_row_id": matches["source_row_id"],
            "fecha": matches["fecha"],
            "tourney_id": matches["tourney_id"],
            "match_num": matches["match_num"],
            "player_id": matches["loser_id"],
            "player_name": matches["loser_name"],
            "opponent_id": matches["winner_id"],
            "opponent_name": matches["winner_name"],
            "is_winner": 0,
            "minutes": matches["minutes"],
            "serve_points": matches["l_svpt"],
            "serve_points_won": matches["l_1stWon"] + matches["l_2ndWon"],
            "return_points": matches["w_svpt"],
            "return_points_won": matches["w_svpt"] - matches["w_1stWon"] - matches["w_2ndWon"],
        }
    )
    player_rows = pd.concat([winner_rows, loser_rows], ignore_index=True)
    numeric_columns = ["minutes", "serve_points", "serve_points_won", "return_points", "return_points_won"]
    for column in numeric_columns:
        player_rows[column] = pd.to_numeric(player_rows[column], errors="coerce")
    player_rows["serve_points_won_pct_match"] = safe_divide(
        player_rows["serve_points_won"],
        player_rows["serve_points"],
    )
    player_rows["return_points_won_pct_match"] = safe_divide(
        player_rows["return_points_won"],
        player_rows["return_points"],
    )
    return player_rows.sort_values(["player_id", "fecha", "tourney_id", "match_num", "source_row_id"]).reset_index(drop=True)


def add_no_leakage_player_features(player_rows: pd.DataFrame) -> pd.DataFrame:
    featured = player_rows.copy()
    grouped = featured.groupby("player_id", group_keys=False)

    featured["serve_points_won_pct_last20"] = grouped["serve_points_won_pct_match"].transform(
        lambda series: series.shift(1).rolling(ROLLING_WINDOW, min_periods=3).mean()
    )
    featured["return_points_won_pct_last20"] = grouped["return_points_won_pct_match"].transform(
        lambda series: series.shift(1).rolling(ROLLING_WINDOW, min_periods=3).mean()
    )
    featured["dias_desde_ultimo_partido"] = grouped["fecha"].diff().dt.days
    featured["returning_from_injury"] = (featured["dias_desde_ultimo_partido"] > 45).astype(int)
    featured["minutos_acumulados_torneo"] = (
        featured.groupby(["player_id", "tourney_id"], group_keys=False)["minutes"]
        .transform(lambda series: series.shift(1).fillna(0).cumsum())
    )
    return featured


def merge_player_features(matches: pd.DataFrame, player_features: pd.DataFrame) -> pd.DataFrame:
    feature_columns = [
        "source_row_id",
        "is_winner",
        "serve_points_won_pct_last20",
        "return_points_won_pct_last20",
        "dias_desde_ultimo_partido",
        "returning_from_injury",
        "minutos_acumulados_torneo",
    ]
    winner_features = (
        player_features.loc[player_features["is_winner"].eq(1), feature_columns]
        .drop(columns=["is_winner"])
        .rename(
            columns={
                "serve_points_won_pct_last20": "jugador_a_serve_points_won_pct_last20",
                "return_points_won_pct_last20": "jugador_a_return_points_won_pct_last20",
                "dias_desde_ultimo_partido": "jugador_a_dias_desde_ultimo_partido",
                "returning_from_injury": "jugador_a_returning_from_injury",
                "minutos_acumulados_torneo": "jugador_a_minutos_acumulados_torneo",
            }
        )
    )
    loser_features = (
        player_features.loc[player_features["is_winner"].eq(0), feature_columns]
        .drop(columns=["is_winner"])
        .rename(
            columns={
                "serve_points_won_pct_last20": "jugador_b_serve_points_won_pct_last20",
                "return_points_won_pct_last20": "jugador_b_return_points_won_pct_last20",
                "dias_desde_ultimo_partido": "jugador_b_dias_desde_ultimo_partido",
                "returning_from_injury": "jugador_b_returning_from_injury",
                "minutos_acumulados_torneo": "jugador_b_minutos_acumulados_torneo",
            }
        )
    )

    output = matches.merge(winner_features, on="source_row_id", how="left")
    output = output.merge(loser_features, on="source_row_id", how="left")
    output = output.rename(
        columns={
            "winner_name": "jugador_a",
            "loser_name": "jugador_b",
        }
    )
    keep_columns = [
        "fecha",
        "tourney_id",
        "tourney_name",
        "surface",
        "best_of",
        "round",
        "jugador_a",
        "jugador_b",
        "total_juegos",
        "diferencia_juegos",
        "jugador_a_serve_points_won_pct_last20",
        "jugador_a_return_points_won_pct_last20",
        "jugador_a_dias_desde_ultimo_partido",
        "jugador_a_returning_from_injury",
        "jugador_a_minutos_acumulados_torneo",
        "jugador_b_serve_points_won_pct_last20",
        "jugador_b_return_points_won_pct_last20",
        "jugador_b_dias_desde_ultimo_partido",
        "jugador_b_returning_from_injury",
        "jugador_b_minutos_acumulados_torneo",
    ]
    output = output[keep_columns].copy()
    output["fecha"] = output["fecha"].dt.date
    return output.sort_values(["fecha", "tourney_id", "round", "jugador_a", "jugador_b"]).reset_index(drop=True)


def build_advanced_features(raw_dir: Path = RAW_DIR, output_path: Path = OUTPUT_PATH) -> pd.DataFrame:
    matches = load_raw_matches(raw_dir)
    player_rows = build_player_match_rows(matches)
    player_features = add_no_leakage_player_features(player_rows)
    output = merge_player_features(matches, player_features)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output.to_csv(output_path, index=False)
    return output


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Build no-leakage advanced ATP features from Sackmann raw CSVs.")
    parser.add_argument("--raw-dir", type=Path, default=RAW_DIR)
    parser.add_argument("--output", type=Path, default=OUTPUT_PATH)
    return parser.parse_args()


def main() -> None:
    args = parse_args()
    frame = build_advanced_features(raw_dir=args.raw_dir, output_path=args.output)
    print(f"Saved {len(frame)} rows to {args.output}")
    frame.info()
    null_rates = frame.isna().mean().sort_values(ascending=False)
    print("\nNull rates:")
    print(null_rates.to_string())


if __name__ == "__main__":
    main()

class ApiFootballETLError(RuntimeError):
    """Raised when the API-Football ETL flow fails."""


@dataclass(frozen=True)
class FixtureQuery:
    league_id: int
    season: int
    last_n: int = 100
    timezone: str = "Europe/Madrid"

    def validate(self) -> None:
        if self.league_id <= 0:
            raise ApiFootballETLError("league_id must be a positive integer.")

        if self.season < 2000:
            raise ApiFootballETLError("season must be a valid starting year.")

        if self.last_n <= 0:
            raise ApiFootballETLError("last_n must be greater than 0.")


class ApiFootballETL:
    """
    Extract and normalize historical fixtures and statistics from API-Football.
    """

    def __init__(self, session: Session | None = None, mock_dir: Path | None = None) -> None:
        self._mock_dir = mock_dir
        self._session = session or requests.Session()

        if self._mock_dir is not None:
            self._base_url = ""
            self._timeout = 30.0
            return

        settings = get_settings()
        self._base_url = settings.api_football_base_url.rstrip("/")
        self._timeout = settings.api_football_timeout_seconds
        self._session.headers.update(
            {
                "X-RapidAPI-Key": settings.api_football_key,
                "X-RapidAPI-Host": settings.api_football_host,
            }
        )

    def fetch_last_league_matches(
        self,
        league_id: int,
        season: int,
        last_n: int = 100,
        timezone: str = "Europe/Madrid",
    ) -> pd.DataFrame:
        query = FixtureQuery(
            league_id=league_id,
            season=season,
            last_n=last_n,
            timezone=timezone,
        )
        query.validate()

        payload = self._get(
            endpoint="fixtures",
            params={
                "league": query.league_id,
                "season": query.season,
                "timezone": query.timezone,
            },
        )
        fixtures = self._extract_response_items(payload=payload)
        cleaned_df = self._clean_fixtures(
            fixtures=fixtures,
            query=query,
            finished_only=True,
        )

        if cleaned_df.empty:
            raise ApiFootballETLError(
                "No finished fixtures were returned for the selected league and season."
            )

        return cleaned_df

    def fetch_fixtures_by_date(
        self,
        fixture_date: str,
        league_id: int | None = None,
        season: int | None = None,
        timezone: str = "Europe/Madrid",
    ) -> pd.DataFrame:
        params: dict[str, Any] = {
            "date": fixture_date,
            "timezone": timezone,
        }
        if league_id is not None:
            params["league"] = league_id
        if season is not None:
            params["season"] = season

        payload = self._get(
            endpoint="fixtures",
            params=params,
        )
        fixtures = self._extract_response_items(payload=payload)
        return self._clean_fixtures(
            fixtures=fixtures,
            query=None,
            finished_only=False,
        )

    def fetch_fixture_statistics(self, fixture_id: int) -> pd.DataFrame:
        if fixture_id <= 0:
            raise ApiFootballETLError("fixture_id must be a positive integer.")

        payload = self._get(
            endpoint="fixtures/statistics",
            params={"fixture": fixture_id},
        )
        stats_items = self._extract_response_items(payload=payload)

        if not stats_items:
            return pd.DataFrame()

        return self._clean_fixture_statistics(fixture_id=fixture_id, stats_items=stats_items)

    def _get(self, endpoint: str, params: dict[str, Any]) -> dict[str, Any]:
        mock_payload = self._load_mock_payload(endpoint=endpoint, params=params)
        if mock_payload is not None:
            self._validate_api_payload(payload=mock_payload)
            return mock_payload

        url = f"{self._base_url}/{endpoint}"

        try:
            response = self._session.get(url, params=params, timeout=self._timeout)
            response.raise_for_status()
        except RequestException as exc:
            raise ApiFootballETLError(
                f"HTTP request to API-Football failed for endpoint={endpoint} params={params}."
            ) from exc

        payload = self._parse_json_response(response=response)
        self._validate_api_payload(payload=payload)
        return payload

    def _load_mock_payload(self, endpoint: str, params: dict[str, Any]) -> dict[str, Any] | None:
        if self._mock_dir is None:
            return None

        api_mock_dir = self._mock_dir / "api_football"
        if endpoint == "fixtures/statistics":
            file_name = f"fixtures_statistics_{params.get('fixture')}.json"
        elif endpoint == "fixtures" and "date" in params:
            file_name = (
                f"fixtures_date_{params.get('date')}_league_{params.get('league', 'all')}"
                f"_season_{params.get('season', 'all')}.json"
            )
        elif endpoint == "fixtures":
            file_name = (
                f"fixtures_league_{params.get('league', 'all')}"
                f"_season_{params.get('season', 'all')}.json"
            )
        else:
            return None

        file_path = api_mock_dir / file_name
        if not file_path.exists():
            raise ApiFootballETLError(f"Mock file not found: {file_path}")

        try:
            return json.loads(file_path.read_text(encoding="utf-8"))
        except json.JSONDecodeError as exc:
            raise ApiFootballETLError(f"Invalid JSON in mock file: {file_path}") from exc

    @staticmethod
    def _parse_json_response(response: Response) -> dict[str, Any]:
        try:
            payload = response.json()
        except ValueError as exc:
            raise ApiFootballETLError(
                "API-Football returned a non-JSON response."
            ) from exc

        if not isinstance(payload, dict):
            raise ApiFootballETLError(
                "API-Football response JSON is not an object."
            )

        return payload

    @staticmethod
    def _validate_api_payload(payload: dict[str, Any]) -> None:
        errors = payload.get("errors")
        response_data = payload.get("response")

        if errors:
            raise ApiFootballETLError(
                f"API-Football reported errors: {errors}"
            )

        if response_data is None:
            raise ApiFootballETLError(
                "API-Football response does not contain a 'response' field."
            )

        if not isinstance(response_data, list):
            raise ApiFootballETLError(
                "API-Football 'response' field is not a list."
            )

    @staticmethod
    def _extract_response_items(payload: dict[str, Any]) -> list[dict[str, Any]]:
        response_items = payload.get("response", [])

        return [
            item for item in response_items
            if isinstance(item, dict)
        ]

    def _clean_fixtures(
        self,
        fixtures: list[dict[str, Any]],
        query: FixtureQuery | None,
        finished_only: bool,
    ) -> pd.DataFrame:
        if not fixtures:
            return pd.DataFrame()

        frame = pd.json_normalize(fixtures, sep="_")

        rename_map = {
            "fixture_id": "fixture_id",
            "fixture_referee": "referee",
            "fixture_timezone": "fixture_timezone",
            "fixture_date": "match_date",
            "fixture_timestamp": "match_timestamp",
            "fixture_periods_first": "first_half_timestamp",
            "fixture_periods_second": "second_half_timestamp",
            "fixture_venue_id": "venue_id",
            "fixture_venue_name": "venue_name",
            "fixture_venue_city": "venue_city",
            "fixture_status_long": "status_long",
            "fixture_status_short": "status_short",
            "fixture_status_elapsed": "status_elapsed_minutes",
            "league_id": "league_id",
            "league_name": "league_name",
            "league_country": "league_country",
            "league_logo": "league_logo",
            "league_flag": "league_flag",
            "league_season": "season",
            "league_round": "round",
            "teams_home_id": "home_team_id",
            "teams_home_name": "home_team_name",
            "teams_home_logo": "home_team_logo",
            "teams_home_winner": "home_team_winner",
            "teams_away_id": "away_team_id",
            "teams_away_name": "away_team_name",
            "teams_away_logo": "away_team_logo",
            "teams_away_winner": "away_team_winner",
            "goals_home": "home_goals",
            "goals_away": "away_goals",
            "score_halftime_home": "halftime_home_goals",
            "score_halftime_away": "halftime_away_goals",
            "score_fulltime_home": "fulltime_home_goals",
            "score_fulltime_away": "fulltime_away_goals",
            "score_extratime_home": "extratime_home_goals",
            "score_extratime_away": "extratime_away_goals",
            "score_penalty_home": "penalty_home_goals",
            "score_penalty_away": "penalty_away_goals",
        }

        available_columns = {
            original: renamed
            for original, renamed in rename_map.items()
            if original in frame.columns
        }
        cleaned = frame.rename(columns=available_columns)
        cleaned = cleaned.loc[:, list(available_columns.values())].copy()

        if finished_only and "status_short" in cleaned.columns:
            cleaned = cleaned.loc[cleaned["status_short"].isin(FINISHED_STATUSES)].copy()

        if "match_date" in cleaned.columns:
            cleaned["match_date"] = pd.to_datetime(
                cleaned["match_date"],
                utc=True,
                errors="coerce",
            )

        integer_like_columns = [
            "fixture_id",
            "league_id",
            "season",
            "venue_id",
            "home_team_id",
            "away_team_id",
            "home_goals",
            "away_goals",
            "halftime_home_goals",
            "halftime_away_goals",
            "fulltime_home_goals",
            "fulltime_away_goals",
            "extratime_home_goals",
            "extratime_away_goals",
            "penalty_home_goals",
            "penalty_away_goals",
            "status_elapsed_minutes",
            "first_half_timestamp",
            "second_half_timestamp",
            "match_timestamp",
        ]
        present_integer_like = [
            column for column in integer_like_columns if column in cleaned.columns
        ]
        for column in present_integer_like:
            cleaned[column] = pd.to_numeric(cleaned[column], errors="coerce")

        for column in ("home_team_winner", "away_team_winner"):
            if column in cleaned.columns:
                cleaned[column] = cleaned[column].astype("boolean")

        cleaned = cleaned.sort_values(
            by=["match_date", "fixture_id"],
            ascending=[False, False],
            na_position="last",
        ).head(query.last_n)
        cleaned = cleaned.reset_index(drop=True)

        if query is not None:
            LOGGER.info(
                "Fetched %s cleaned fixtures for league=%s season=%s.",
                len(cleaned),
                query.league_id,
                query.season,
            )
        else:
            LOGGER.info("Fetched %s cleaned fixtures for date query.", len(cleaned))

        return cleaned

    def _clean_fixture_statistics(
        self,
        fixture_id: int,
        stats_items: list[dict[str, Any]],
    ) -> pd.DataFrame:
        rows: list[dict[str, Any]] = []

        for item in stats_items:
            team = item.get("team", {})
            statistics = item.get("statistics", [])

            row: dict[str, Any] = {
                "fixture_id": fixture_id,
                "team_id": team.get("id"),
                "team_name": team.get("name"),
                "team_logo": team.get("logo"),
            }

            for stat in statistics:
                stat_type = stat.get("type")
                stat_value = stat.get("value")

                if not stat_type:
                    continue

                normalized_name = STAT_KEY_MAP.get(stat_type)
                if not normalized_name:
                    continue

                row[normalized_name] = self._coerce_stat_value(stat_value)

            rows.append(row)

        if not rows:
            return pd.DataFrame()

        frame = pd.DataFrame(rows)
        numeric_columns = [column for column in frame.columns if column not in {"team_name", "team_logo"}]
        for column in numeric_columns:
            if column in {"fixture_id", "team_id"}:
                frame[column] = pd.to_numeric(frame[column], errors="coerce")
                continue

            frame[column] = pd.to_numeric(frame[column], errors="coerce")

        return frame

    @staticmethod
    def _coerce_stat_value(value: Any) -> float | int | None:
        if value is None:
            return None

        if isinstance(value, (int, float)):
            return value

        if isinstance(value, str):
            cleaned = value.strip().replace("%", "")
            if cleaned == "":
                return None

            try:
                numeric_value = float(cleaned)
            except ValueError:
                return None

            if numeric_value.is_integer():
                return int(numeric_value)

            return numeric_value

        return None
